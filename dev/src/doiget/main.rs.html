14/// `--color` value (#211 / CONFIG.md §5).
15///
16/// Honored by future ANSI emitters; `Auto` decides per-stderr-TTY at the
17/// emission site. The `NO_COLOR` cross-tool convention
18/// (<https://no-color.org/>) is honored by the *consumer-side* resolver,
19/// not at this write boundary: `apply_global_overrides` writes
20/// `DOIGET_COLOR` from the flag value unconditionally, and a future
21/// ANSI emitter MUST check `NO_COLOR` (present and non-empty, regardless
22/// of value) before consulting `DOIGET_COLOR`. The split keeps the
23/// write boundary stupid-simple and centralises the NO_COLOR
24/// precedence in the emitter's own resolver where it is closer to the
25/// actual rendering site.
26///
27/// This slice ships the *surface*: the flag is accepted and the resolved
28/// value is exposed via `DOIGET_COLOR` so any future ANSI consumer can
29/// read it through the same env-driven layer as the rest of the config.
30/// No consumer currently emits ANSI to be gated.
31#[derive(Debug, Clone, Copy, PartialEq, Eq, ValueEnum)]
32#[clap(rename_all = "lower")]
33enum OutputColor {
34    /// Color when stderr is a terminal, monochrome otherwise. This is
35    /// also the default when `--color` is not given; explicit
36    /// `--color auto` is therefore a no-op vs the default unless
37    /// `NO_COLOR` is set, in which case the *consumer-side* resolver
38    /// still forces monochrome regardless of `--color`.
39    Auto,
40    /// Always emit ANSI, even on a pipe. Note that `NO_COLOR` (present
41    /// + non-empty) still wins at the consumer-side resolver.
42    Always,
43    /// Never emit ANSI, even on a terminal.
44    Never,
45}
46
47impl OutputColor {
48    /// String representation written to `DOIGET_COLOR`.
49    ///
50    /// The match arms intentionally mirror the `#[clap(rename_all =
51    /// "lower")]` parser-side spelling so a round trip through clap →
52    /// `as_env_value` → env → future consumer is byte-identical.
53    /// The pair is pinned by
54    /// `tests::output_color_env_strings_match_clap_parser_side`, which
55    /// rejects each variant's string against clap's own
56    /// `ValueEnum::to_possible_value().get_name()` — if `rename_all`
57    /// is changed (or a variant renamed), the test fires before any
58    /// env consumer observes the drift.
59    fn as_env_value(self) -> &'static str {
60        match self {
61            OutputColor::Auto => "auto",
62            OutputColor::Always => "always",
63            OutputColor::Never => "never",
64        }
65    }
66}

68/// Custom value parser for path-shaped CLI arguments.
69///
70/// Rejects empty strings and NUL bytes at parse time. NUL would
71/// otherwise panic later in `std::env::set_var` (silent abort under
72/// `panic = "abort"`); the empty-string case was a documented silent
73/// failure path from the review pass (`apply_global_overrides`
74/// would have written `DOIGET_*=""` which downstream resolvers treat
75/// differently from an unset key).
76///
77/// Returning `Utf8PathBuf` honors the workspace `clippy.toml`
78/// disallowed-types policy (`std::path::PathBuf` is banned) and
79/// validates UTF-8 at the parse boundary instead of on first use.
80fn parse_utf8_path(raw: &str) -> Result<Utf8PathBuf, String> {
81    if raw.is_empty() {
82        return Err("path must not be empty".to_string());
83    }
84    if raw.contains('\0') {
85        return Err("path must not contain NUL bytes".to_string());
86    }
87    Ok(Utf8PathBuf::from(raw))
88}

215impl Cli {
216    /// Collapse the `progress` / `no_progress` bool pair into the
217    /// three-state value it actually represents (review pass I7).
218    ///
219    /// Returns `Some(true)` for `--progress`, `Some(false)` for
220    /// `--no-progress`, and `None` when neither flag was given.
221    /// The `(true, true)` case is unreachable thanks to clap's
222    /// `conflicts_with`, but the helper is unconditional so callers
223    /// don't need to remember which flag the if-arm prioritises.
224    fn progress_choice(&self) -> Option<bool> {
225        match (self.progress, self.no_progress) {
226            (true, _) => Some(true),
227            (_, true) => Some(false),
228            _ => None,
229        }
230    }
231}

616#[tokio::main]
617async fn main() -> anyhow::Result<()> {
618    // Logging — strictly to stderr. See docs/SECURITY.md §3 / ADR-0001.
619    tracing_subscriber::fmt()
620        .with_writer(std::io::stderr)
621        .with_env_filter(tracing_subscriber::EnvFilter::from_default_env())
622        .init();
623
624    let cli = Cli::parse();
625
626    let result: anyhow::Result<()> = run_dispatch(cli).await;
627
628    // Issue #119: a `CliExit` carries a `docs/ERRORS.md` §4 process
629    // exit code and means the human-readable `error[CODE]:` line was
630    // ALREADY printed to stderr by the command. `main` owns the actual
631    // process exit (doing it inside the command would kill in-process
632    // integration tests). Every other error keeps the default anyhow
633    // behaviour (Debug chain to stderr, exit 1).
634    match result {
635        Ok(()) => Ok(()),
636        Err(err) => match err.downcast_ref::<doiget_cli::commands::fetch::CliExit>() {
637            Some(doiget_cli::commands::fetch::CliExit(code)) => {
638                std::process::exit(*code);
639            }
640            None => Err(err),
641        },
642    }
643}

645/// Build the `FlagInput` from the three mutually-exclusive global
646/// flags. Clap's `conflicts_with_all` guarantees at most one is set, so
647/// the ordering of the if-arms below is irrelevant to correctness.
648fn flag_input_from(cli: &Cli) -> FlagInput {
649    if let Some(m) = cli.mode {
650        FlagInput::Explicit(m)
651    } else if cli.json {
652        FlagInput::JsonShort
653    } else if cli.quiet {
654        FlagInput::QuietShort
655    } else {
656        FlagInput::None
657    }

660/// Compute the `forced_implicit` mode from the subcommand. Only `serve`
661/// pins a mode (`Mcp`) — CONFIG.md §5 / ADR-0017 / SECURITY.md §3: the
662/// MCP server emits JSON-RPC frames on stdout and a `--mode quiet` /

672/// Apply the four CONFIG.md §5 global flags (#211) by overwriting the
673/// matching process env vars before any command resolver reads them.
674///
675/// This keeps the precedence ladder `flag > env > default` correct
676/// uniformly across every command without threading a new
677/// `ResolvedFlags` value through eleven `run(..)` signatures: the
678/// existing env-driven resolvers (`resolve_store_root`,
679/// `resolve_log_path`, the future `--color` / `--progress`
680/// consumers) keep reading env and will see the flag value when one
681/// is given.
682///
683/// `--color` and `--progress` / `--no-progress` are *surface-only* in
684/// this slice — they are accepted and exposed via `DOIGET_COLOR` /
685/// `DOIGET_PROGRESS` for future consumers, but doiget does not yet
686/// emit ANSI escapes or per-ref progress lines that would respect
687/// them. `NO_COLOR=1` (per <https://no-color.org/>) still wins over
688/// `--color always` because the consumer-side resolver checks
689/// `NO_COLOR` first by the standard convention.
690///
691/// **Thread safety:** `std::env::set_var` is safe on edition 2021 /
692/// Rust 1.86 (the workspace MSRV; cf. `Cargo.toml [workspace.package]
693/// edition`). Under edition 2024 it becomes `unsafe fn` (the
694/// `setenv` POSIX call is documented as non-thread-safe), and the
695/// workspace `-F unsafe-code` lint will need a localized
696/// `#[allow(unsafe_code)]` here when migrating. Tracked under #211
697/// follow-up.
698///
699/// We invoke this from `run_dispatch` (see call site below) before
700/// any `.await` in the function. The `#[tokio::main]` runtime has
701/// already constructed its multi-thread worker pool by that point,
702/// but the workers are parked on the work queue and do not read
703/// `environ`; the active thread is the binary's startup thread. The
704/// concurrent-read soundness condition for `setenv` is therefore met.
705/// If a future change introduces an async background task that reads
706/// `DOIGET_STORE_ROOT` (or any other key this function writes), the
707/// application of overrides must move ahead of the runtime
708/// construction (i.e. above the `#[tokio::main]` boundary).
709///
710/// The function intentionally has no error path: each flag value has
711/// already passed `parse_utf8_path` (empty and NUL rejected) or
712/// clap's `ValueEnum` (closed set of strings), so `set_var` cannot
713/// panic on the value. The function is fire-and-forget by design.
714fn apply_global_overrides(cli: &Cli) {
715    if let Some(v) = cli.store_root.as_deref() {
716        std::env::set_var("DOIGET_STORE_ROOT", v.as_str());
717    }
718    if let Some(v) = cli.log_path.as_deref() {
719        std::env::set_var("DOIGET_LOG_PATH", v.as_str());
720    }
721    if let Some(c) = cli.color {
722        std::env::set_var("DOIGET_COLOR", c.as_env_value());
723    }
724    // `--progress` / `--no-progress` collapse into a typed
725    // three-state via `Cli::progress_choice` (review pass I7). When
726    // neither is given we leave `DOIGET_PROGRESS` untouched so a
727    // user-set env value survives and the future consumer can fall
728    // back to its own TTY-based default.
729    if let Some(on) = cli.progress_choice() {
730        std::env::set_var("DOIGET_PROGRESS", if on { "1" } else { "0" });
731    }
732}

734async fn run_dispatch(cli: Cli) -> anyhow::Result<()> {
735    // CONFIG.md §5 / #211: apply global path / display flags by
736    // overwriting their env counterparts BEFORE any resolver reads
737    // them. The precedence
738    //
739    //   flag (this fn) > env > default
740    //
741    // is implemented by `apply_global_overrides`'s
742    // `--flag-wins-by-setting-env-first` pattern: when a flag is
743    // supplied, its value replaces the env value; when no flag is
744    // supplied, the env value (or the resolver's default) stands.
745    apply_global_overrides(&cli);
746
747    // Resolve the effective output mode ONCE per invocation per ADR-0017.
748    // The pure resolver lives in `commands::output`; this site is the
749    // single I/O-touching layer that reads env + probes the TTY.
750    // ADR-0017 Amendment 1: the resolver returns ResolvedOutput
751    // carrying mode + quiet_was_explicit; informational commands receive
752    // `out.mode` (their existing OutputMode signature), artifact
753    // commands additionally consume `out.quiet_was_explicit` so the
754    // non-TTY fallback to Quiet does NOT silence them (#219 / #220).
755    let out = output::resolve(
756        forced_implicit_for(&cli.command),
757        flag_input_from(&cli),
758        std::env::var("DOIGET_MODE").ok().as_deref(),
759        output::stdout_is_tty(),
760    );
761    let mode = out.mode;
762
763    match cli.command {
764        None => {
765            anyhow::bail!("no subcommand. Run `doiget --help` for available commands.");
766        }
767        // Phase 1 subcommands. All command modules live in the library half
768        // of this crate (see `src/lib.rs`) so integration tests can drive them
769        // in-process.
770        //
771        // Each command receives the resolved `mode`. Per-mode behaviour
772        // (Quiet stdout suppression, Json bodies for human-table
773        // commands) is tracked in follow-up issues #203 / #204 / #205;
774        // this PR only wires the threading and the `serve→Mcp` invariant.
775        Some(Command::AuditLog { verify }) => doiget_cli::commands::audit_log::run(verify, mode),
776        Some(Command::Provenance { action }) => match action {
777            ProvenanceAction::Migrate { dry_run } => {

785        }) => {

787                .await
788        }
789        Some(Command::Info { ref_ }) => {
790            doiget_cli::commands::info::run(ref_, mode, out.quiet_was_explicit)
791        }
792        Some(Command::ListRecent { limit, missing_pdf }) => {
793            doiget_cli::commands::list_recent::run(limit, missing_pdf, mode, out.quiet_was_explicit)
794        }
795        Some(Command::Search {

807            author,

809            publisher,
810            sort,
811        }) => {

815                to_year,
816                oa_only,
817                min_citations,
818                min_fwci,

820                author,
821                venue,
822                publisher,
823                sort,
824            };
825            doiget_cli::commands::search::run(query, local, tag, ext, mode, out.quiet_was_explicit)
826                .await
827        }
828        Some(Command::Version { check }) => doiget_cli::commands::version::run(check, mode).await,
829        Some(Command::Verify {
830            path,
831            format,
832            strict,
833        }) => doiget_cli::commands::verify::run(path, format, strict, mode).await,
834        Some(Command::Lint { path, strict }) => doiget_cli::commands::lint::run(path, strict, mode),

946        // tracing output will not collide with JSON-RPC frames on stdout.

952        Some(Command::Serve) => {

964        // is what closes the LLM cold-boot deadlock reported in
965        // #219 / #220.

967            let cli_cmd = <Cli as clap::CommandFactory>::command();

969        }
970        // Phase 4 / Slice 16. Feature-gated to keep default release
971        // binaries free of the OpenAlex-only citation walker.
972        #[cfg(feature = "citation")]
973        Some(Command::Graph {
974            ref_,
975            depth,
976            total,
977            per_paper,
978        }) => doiget_cli::commands::graph::run(ref_, depth, total, per_paper, mode).await,
979    }
