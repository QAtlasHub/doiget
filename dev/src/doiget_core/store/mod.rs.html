1//! Filesystem-backed metadata store.
2//!
3//! Binding spec: [`docs/STORE.md`](../../../../docs/STORE.md) (NORMATIVE shared
4//! spec for layout, schema, lock protocol, atomic write, normalization).
5//! Public API surface: `docs/PUBLIC_API.md` §2 (Store trait), §3 (Metadata).
6//!
7//! ## Entry points
8//!
9//! - [`Store`] — the trait surface implementations expose.
10//! - [`FsStore`] — filesystem-backed implementation rooted at a configurable
11//!   directory (default `./papers`, under the cwd; ADR-0036).
12//! - [`Metadata`] / [`DoigetExtension`] — the on-disk schema, mirrored from
13//!   `docs/STORE.md` §2.
14//!
15//! ## Cross-tool coexistence
16//!
17//! The store can be shared between doiget and BiblioFetch.jl when both are
18//! pointed at the same root (they no longer co-locate by default — ADR-0036).
19//! Both tools follow the lock protocol in `docs/STORE.md` §4 and the
20//! atomic-write sequence in §5. Per §6, doiget MUST NOT overwrite reserved
21//! top-level fields previously written by another tool — see [`FsStore::write`].

41/// Brief summary of a stored entry; returned by
42/// [`Store::list_recent`] / [`Store::search`].
43///
44/// `non_exhaustive` so adding new summary fields (e.g. `doi`, `authors`) in a
45/// later revision is non-breaking. Pattern-match with a wildcard arm.
46///
47/// `Serialize` enables `list-recent --mode json` / `search --mode json`
48/// (#204) — the wire form is the obvious field-name JSON: `{"safekey":
49/// "...", "title": "...", "year": 2024, "fetched_at": "2026-05-20T…Z"}`,
50/// with `null` for absent optionals.
51///
52/// # Wire-format stability (post-#208 self-review §1)
53///
54/// Once a release ships with the \[`Serialize`\] derive, the field
55/// **names** below become part of the public API: a downstream consumer
56/// (CLI agent, MCP tool, BiblioFetch.jl, third-party script) MAY bind
57/// to them. Renaming a field is then a semver minor bump and warrants
58/// a CHANGELOG \[BREAKING\] note. Adding new fields is still safe
59/// (per `#[non_exhaustive]`).
60#[derive(Debug, Clone, Serialize)]
61#[non_exhaustive]
62pub struct EntryInfo {
63    /// The safekey of the entry. See `docs/SAFEKEY.md`.
64    pub safekey: Safekey,
65    /// Title from the entry's reserved `title` field.
66    pub title: String,
67    /// Year, if any, from the entry's reserved `year` field.
68    pub year: Option<i32>,
69    /// `fetched_at` from the `[doiget]` table, if any.
70    pub fetched_at: Option<chrono::DateTime<chrono::Utc>>,
71    /// `size_bytes` from the `[doiget]` table: the size of the stored PDF,
72    /// `0` for a metadata-only entry, `None` when the entry has no
73    /// `[doiget]` table at all.
74    ///
75    /// #481: without it the inventory commands could not tell a fetched
76    /// paper from a metadata-only stub. Every other surface could -- the
77    /// fetch itself failed loudly, the TOML omits `pdf_path`, the
78    /// provenance log carries an `err` row, `doiget info` shows
79    /// `size_bytes = 0` -- and the one command that answers "what do I
80    /// have?" without knowing the ref in advance was the one that dropped
81    /// it. Fifty refs with ten blocked listed as fifty identical rows.
82    pub size_bytes: Option<u64>,
83}
84
85impl EntryInfo {
86    /// Whether a PDF was actually stored for this entry.
87    ///
88    /// `false` for a metadata-only entry (`size_bytes == 0`) and for one
89    /// with no `[doiget]` table. Deliberately not "is this entry useful" --
90    /// a metadata-only entry is a legitimate result, it is just a different
91    /// one, and #118 is the standing rule that the two must not be
92    /// presented alike.
93    #[must_use]
94    pub fn has_pdf(&self) -> bool {
95        self.size_bytes.is_some_and(|n| n > 0)
96    }
97}

99/// Errors emitted by [`Store`] implementations.
100#[derive(Debug, Error)]
101#[non_exhaustive]
102pub enum StoreError {
103    /// Underlying I/O failure.
104    #[error("io error: {0}")]
105    Io(#[from] std::io::Error),
106    /// Malformed TOML or schema mismatch on read.
107    #[error("toml deserialize error: {0}")]
108    Deserialize(#[from] toml::de::Error),
109    /// Failed to serialize a [`Metadata`] to TOML.
110    #[error("toml serialize error: {0}")]
111    Serialize(#[from] toml::ser::Error),
112    /// Could not acquire the advisory `flock` within the 5 s budget named in
113    /// `docs/STORE.md` §4.
114    #[error("flock timeout (5s) on {path}")]
115    LockTimeout {
116        /// The lock-file path that was contended.
117        path: camino::Utf8PathBuf,
118    },
119    /// The on-disk `schema_version` is a future major; per `docs/STORE.md` §3
120    /// the entry is read-only for this build.
121    #[error("schema_version too new: {theirs} > {ours}; entry is read-only")]
122    SchemaTooNew {
123        /// Schema version observed on disk.
124        theirs: String,
125        /// Schema version this build supports.
126        ours: String,
127    },
128    /// A reserved field that the spec marks as required is missing.
129    #[error("required field missing: {field}")]
130    MissingField {
131        /// The name of the missing reserved field.
132        field: &'static str,
133    },
134    /// The supplied [`Safekey`] resolves to a path outside the store root.
135    /// Defense-in-depth check; `Safekey` construction already enforces the
136    /// `[A-Za-z0-9._-]`-only charset per `docs/SAFEKEY.md`.
137    #[error("path is outside the store root: {path}")]
138    PathTraversal {
139        /// The offending resolved path.
140        path: camino::Utf8PathBuf,
141    },
142}

144/// Who is authoritative for the user-authored `[doiget]` fields
145/// (`tags`, `collections`, `annotation`) on a write.
146///
147/// A fetch never authors them: every `DoigetExtension` the orchestrator
148/// builds hard-codes `Vec::new()` / `None`. Letting that win silently
149/// discarded a user's tags on any re-fetch, which is the loss ADR-0056
150/// closed for `oa_status` / `license` and left open here.
151///
152/// The distinction has to be explicit rather than "is the incoming value
153/// empty", because `doiget tag --remove` and `doiget annotate --clear`
154/// legitimately mean the empty value.
155#[derive(Debug, Clone, Copy, PartialEq, Eq)]
156#[non_exhaustive]
157pub enum UserFields {
158    /// The caller did not author them; keep whatever is on disk. Fetches.
159    Preserve,
160    /// The caller means exactly what it wrote, empty included. `doiget tag`,
161    /// `doiget annotate`, and their MCP equivalents.
162    Authored,
163}

165/// Filesystem-shaped metadata store, semver-locked per `docs/PUBLIC_API.md`
166/// §2.
167///
168/// Implementations are responsible for honoring:
169///
170/// - `docs/STORE.md` §4 lock protocol (advisory `flock` on
171///   `<safekey>.toml.lock` with a 5 s timeout).
172/// - `docs/STORE.md` §5 atomic-write sequence (`tmp` → fsync → rename →
173///   fsync parent).
174/// - `docs/STORE.md` §6 doiget write discipline: never overwrite reserved
175///   top-level fields previously written by another tool.
176/// - `docs/STORE.md` §7 TOML normalization (alphabetical key order, `\n`
177///   line endings, trailing newline).
178pub trait Store: Send + Sync {
179    /// Read the entry keyed by `key`.
180    ///
181    /// Returns `Ok(None)` if no entry exists. Returns `Err` on I/O failure,
182    /// malformed TOML, or unrecoverable schema mismatch (e.g. future major).
183    fn read(&self, key: &Safekey) -> Result<Option<Metadata>, StoreError>;
184
185    /// Write or update the entry keyed by `key`.
186    ///
187    /// If `pdf` is `Some`, the file at that path is copied to
188    /// `<root>/<safekey>.pdf` via the same atomic-rename dance as the
189    /// metadata file. The caller is responsible for emitting the
190    /// `event=store_write` provenance row (see `docs/PROVENANCE_LOG.md` §3).
191    /// Write a fetch result. User-authored `[doiget]` fields already on disk
192    /// are preserved ([`UserFields::Preserve`]) -- a fetch does not author
193    /// them, and silently dropping them is data loss.
194    fn write(&self, key: &Safekey, m: &Metadata, pdf: Option<&Utf8Path>) -> Result<(), StoreError>;
195
196    /// Write on behalf of a caller that DID author the user fields, so an
197    /// empty `tags` / `collections` or a `None` annotation means exactly that
198    /// ([`UserFields::Authored`]). `doiget tag` / `doiget annotate` only.
199    fn write_user_authored(
200        &self,
201        key: &Safekey,
202        m: &Metadata,
203        pdf: Option<&Utf8Path>,
204    ) -> Result<(), StoreError>;
205
206    /// Return up to `limit` entries, most-recent first by `[doiget].fetched_at`.
207    fn list_recent(&self, limit: usize) -> Result<Vec<EntryInfo>, StoreError>;
208
209    /// Return up to `limit` entries whose title / authors / venue / publisher
210    /// case-insensitively contain `query`.
211    fn search(&self, query: &str, limit: usize) -> Result<Vec<EntryInfo>, StoreError>;
212}
