1//! Bibliography input adapters per ADR-0030.
2//!
3//! Parses three input shapes into an iterator of `Ref`s with optional
4//! `entry_key` provenance back to the source bibliography:
5//!
6//! - **Plain refs**: one `doi:…` / `arxiv:…` / bare-DOI / bare-arXiv id
7//!   per line, with `#`-prefixed comments and blank lines tolerated.
8//!   The existing `doiget batch <refs.txt>` shape.
9//! - **CSL-JSON**: a JSON array of entries with `id` (citation key),
10//!   `DOI`, and optionally `archivePrefix = "arXiv"` + `eprint`
11//!   fields. Parsed via the workspace's existing `serde_json` — no
12//!   new dependency.
13//! - **BibTeX / BibLaTeX (.bib)**: parsed via the `biblatex` crate
14//!   (ADR-0030 D2). One `@entrytype{KEY, …}` per entry; the `doi`
15//!   field is preferred, falling back to an arXiv `eprint`.
16//!
17//! Identifier-pick priority per ADR-0030 D3: `doi` > `arxiv` > `pmid`
18//! (PMID adapter parking until the `Ref::Pmid` variant lands in a
19//! later slice; current code carries the rule through without
20//! producing a `Pmid` ref).
21//!
22//! Parse-error policy per ADR-0030 D5: a single entry's failure is
23//! captured per-entry and does NOT abort the whole batch. The caller
24//! decides whether to skip-and-warn (default) or fail-closed
25//! (`--strict`).
26

33/// One successfully-parsed bibliography entry.
34///
35/// `entry_key` echoes the source bibliography's citation key
36/// (BibTeX `@article{KEY,…}` / CSL-JSON `"id"`) so downstream
37/// automation can bridge the fetch outcome back to the originating
38/// reference — the load-bearing field for the Zotero / Mendeley
39/// "attach fetched PDF to this reference" workflow per ADR-0030 §6.
40#[derive(Debug, Clone, PartialEq, Eq)]
41#[non_exhaustive]
42pub struct ParsedEntry {
43    /// The identifier the adapter chose for this entry (`Ref::Doi` /
44    /// `Ref::Arxiv`).
45    pub ref_: Ref,
46    /// The source bibliography's citation key, when one is available.
47    /// `None` for plain-refs input (no key concept) and for any
48    /// future input shape that lacks per-entry keys.
49    pub entry_key: Option<String>,
50}

52/// Why a single bibliography entry failed to produce a `Ref`.
53///
54/// Closed-enum so the failure-class can be exposed at the
55/// `docs/ERRORS.md` §3 INVALID_REF surface without leaking parser
56/// internals.
57#[derive(Debug, Clone, Error, PartialEq, Eq)]
58#[non_exhaustive]
59pub enum ParseError {
60    /// The line did not contain a `doi:` / `arxiv:` / bare-DOI /
61    /// bare-arXiv id — empty (after trimming) or just a comment.
62    /// Plain-refs path filters these out silently; CSL-JSON path
63    /// emits this when an entry has no resolvable identifier.
64    #[error("entry has no DOI / arXiv id (entry_key={entry_key:?})")]
65    NoIdentifier {
66        /// The source bibliography's citation key, when known.
67        entry_key: Option<String>,
68    },
69    /// The entry DOES carry an identifier, and it is one doiget recognises
70    /// and cannot resolve yet (#500).
71    ///
72    /// Distinct from [`Self::NoIdentifier`] because the two send a reader in
73    /// opposite directions. "entry has no DOI / arXiv id" is accurate about
74    /// what the parser did and wrong about the entry: a PubMed-exported
75    /// `.bib` record carrying `pmid = {9659853}` is not deficient, and a user
76    /// who believes it is will go and edit a bibliography that was fine. The
77    /// missing piece is on doiget's side.
78    ///
79    /// Surfaces as `NOT_IMPLEMENTED` rather than `INVALID_REF`: the input is
80    /// valid and the support is absent, and the two carry different advice --
81    /// "wait for a release" versus "correct your input" (ADR-0055).
82    #[error("entry {entry_key:?} is identified only by {kind} {value:?}, which doiget cannot resolve yet (issue #500) -- it is NOT missing an identifier")]
83    UnsupportedIdentifier {
84        /// Human-facing name of the identifier class, e.g. `"PMID"`.
85        kind: &'static str,
86        /// The identifier as written in the entry.
87        value: String,
88        /// The source bibliography's citation key, when known.
89        entry_key: Option<String>,
90    },
91    /// The identifier was present but `Ref::parse` rejected it
92    /// (malformed DOI suffix, invalid arXiv id shape, etc.).
93    #[error(
94        "entry identifier {raw:?} did not parse as a Ref \
95         (entry_key={entry_key:?}): {source}"
96    )]
97    InvalidRef {
98        /// The raw identifier string the parser saw.
99        raw: String,
100        /// The source bibliography's citation key, when known.
101        entry_key: Option<String>,
102        /// The structured `Ref::parse` failure.
103        #[source]
104        source: RefParseError,
105    },
106    /// The whole input did not deserialise — CSL-JSON that is not a
107    /// JSON array, top-level malformed JSON, etc. This is a
108    /// whole-input failure, not a per-entry failure; callers receive
109    /// it as the sole `Err` element of the result iterator.
110    #[error("input did not deserialise as {format}: {message}")]
111    Decode {
112        /// Which parser branch produced the failure (`"csl-json"` /
113        /// `"bibtex"`).
114        format: &'static str,
115        /// `serde_json::Error::to_string()`.
116        message: String,
117    },
118    /// Format requested or detected, but no parser for it is shipped.
119    /// Retained as a forward-compatible variant for input shapes not
120    /// yet implemented (e.g. a future RIS adapter); the `bibtex` and
121    /// `csl-json` paths are both live today.
122    #[error("{format} parsing is not yet implemented")]
123    UnsupportedFormat {
124        /// The format token naming the unsupported shape.
125        format: &'static str,
126    },
127}

129/// The claim [`ParseError::UnsupportedIdentifier`] makes, without the
130/// `entry {entry_key:?}` prefix its `Display` carries.
131///
132/// Callers that put `entry_key` in a field of its own -- the CLI `verify`
133/// row and the MCP `batch_from_bibliography` envelope both do -- would
134/// otherwise say it twice. One definition rather than a copy at each site,
135/// because the copies drifted: two of the three carried a run of joined-line
136/// whitespace into user-facing output before anything asserted the text.
137#[must_use]
138pub fn unsupported_identifier_claim(kind: &str, value: &str) -> String {
139    format!("entry is identified only by {kind} {value:?}, which doiget cannot resolve yet (issue #500); it is NOT missing an identifier")
140}

142/// Input-shape discriminator per ADR-0030 D4.
143///
144/// `Auto` means "detect from path extension and/or content
145/// fingerprint"; the explicit variants name a parser directly and
146/// skip detection.
147#[derive(Debug, Clone, Copy, PartialEq, Eq)]
148#[non_exhaustive]
149pub enum Format {
150    /// Detect from file extension if a path was supplied, else from
151    /// content fingerprint; fall through to [`Format::Refs`].
152    Auto,
153    /// Plain refs — one identifier per line, `#` comments, blanks.
154    Refs,
155    /// CSL-JSON array per <https://citationstyles.org/>.
156    CslJson,
157    /// BibTeX / BibLaTeX, parsed via the `biblatex` crate (ADR-0030 D2).
158    Bibtex,
159}
160
161impl Format {
162    /// Wire token used by the CLI `--format` flag and the MCP tool
163    /// input schema's `format` field per ADR-0030 §6.
164    pub fn as_wire(&self) -> &'static str {
165        match self {
166            Format::Auto => "auto",
167            Format::Refs => "refs",
168            Format::CslJson => "csl-json",
169            Format::Bibtex => "bibtex",
170        }
171    }
172}

174/// Detect the input format per ADR-0030 D4.
175///
176/// Precedence: file extension first (when `path` is `Some`), then
177/// content fingerprint, then fallback to [`Format::Refs`]. The
178/// caller's explicit `--format` flag should short-circuit this
179/// function — it is the slowest of the three precedence rules in the
180/// ADR.
181pub fn detect_format(path: Option<&Utf8Path>, content: &str) -> Format {
182    if let Some(p) = path {
183        let ext = p.extension().unwrap_or_default().to_ascii_lowercase();
184        match ext.as_str() {
185            "bib" | "biblatex" => return Format::Bibtex,
186            "json" | "csl" => return Format::CslJson,
187            _ => {}
188        }
189    }
190    // Content fingerprint: peek the first non-blank, non-comment line.
191    for line in content.lines() {
192        let trimmed = line.trim();
193        if trimmed.is_empty() || trimmed.starts_with('#') {
194            continue;
195        }
196        if trimmed.starts_with('@') {
197            return Format::Bibtex;
198        }
199        if trimmed.starts_with('[') || trimmed.starts_with('{') {
200            return Format::CslJson;
201        }
202        break;
203    }
204    Format::Refs
205}

207/// Parse `text` per `format`, dispatching to the matching shape
208/// parser. `path` is consulted only when `format == Format::Auto` to
209/// drive [`detect_format`].
210///
211/// Returns one element per discovered entry — `Ok` for entries that
212/// produced a `Ref`, `Err` for per-entry failures the caller should
213/// surface as a JSONL `INVALID_REF` line. A whole-input decode
214/// failure ([`ParseError::Decode`]) is returned as a single-element
215/// `Err` so the caller's exit-code path treats it as one parse error
216/// rather than zero.
217pub fn parse_input(
218    text: &str,
219    format: Format,
220    path: Option<&Utf8Path>,
221) -> Vec<Result<ParsedEntry, ParseError>> {
222    let resolved = match format {
223        Format::Auto => detect_format(path, text),
224        other => other,
225    };
226    match resolved {
227        Format::Refs | Format::Auto => parse_plain_refs(text),
228        Format::CslJson => parse_csl_json(text),
229        Format::Bibtex => parse_bibtex(text),
230    }
231}

233/// Parse plain refs — the existing batch input format. One ref per
234/// non-blank, non-comment line. `entry_key` is always `None` for this
235/// shape; plain refs have no citation-key concept.
236pub fn parse_plain_refs(text: &str) -> Vec<Result<ParsedEntry, ParseError>> {
237    let mut out = Vec::new();
238    for raw_line in text.lines() {
239        let line = raw_line.trim();
240        if line.is_empty() || line.starts_with('#') {
241            continue;
242        }
243        out.push(match Ref::parse(line) {
244            Ok(ref_) => Ok(ParsedEntry {
245                ref_,
246                entry_key: None,
247            }),
248            Err(e) => Err(ParseError::InvalidRef {
249                raw: line.to_string(),
250                entry_key: None,
251                source: e,
252            }),
253        });
254    }
255    out
256}

258/// Parse a CSL-JSON document — a JSON array of objects, each with at
259/// least an `id` (citation key) and one of `DOI`, or `archivePrefix`
260/// + `eprint` (arXiv).
261///
262/// Identifier-pick priority per ADR-0030 D3:
263///
264/// 1. `DOI` field (case-sensitive per the CSL-JSON spec but Zotero
265///    sometimes emits `doi` lowercase — we accept both).
266/// 2. `archivePrefix == "arXiv"` (case-insensitive) + `eprint`
267///    (or `note: "arXiv:..."` shape Zotero emits).
268/// 3. (PMID parking — `Ref::Pmid` not yet defined; PMIDs in CSL-JSON
269///    are recorded as parse failures with `NoIdentifier` until the
270///    variant lands.)
271///
272/// `entry_key` is the `id` field verbatim.
273pub fn parse_csl_json(text: &str) -> Vec<Result<ParsedEntry, ParseError>> {
274    let parsed: serde_json::Result<Vec<serde_json::Value>> = serde_json::from_str(text);
275    let entries = match parsed {
276        Ok(arr) => arr,
277        Err(e) => {
278            return vec![Err(ParseError::Decode {
279                format: "csl-json",
280                message: e.to_string(),
281            })]
282        }
283    };
284    let mut out = Vec::with_capacity(entries.len());
285    for entry in entries {
286        // `id` is usually a string in real-world Zotero exports but
287        // the spec allows numeric ids too — stringify either form so
288        // the operator can find the entry in their library.
289        let entry_key = entry.get("id").and_then(|v| {
290            if let Some(s) = v.as_str() {
291                Some(s.to_string())
292            } else if v.is_number() {
293                Some(v.to_string())
294            } else {
295                None
296            }
297        });
298        out.push(parse_csl_entry(&entry, entry_key));
299    }
300    out
301}

303/// Pick the highest-priority identifier on a single CSL-JSON entry
304/// and parse it. Honors ADR-0030 D3 priority.
305fn parse_csl_entry(
306    entry: &serde_json::Value,
307    entry_key: Option<String>,
308) -> Result<ParsedEntry, ParseError> {
309    // Priority 1: DOI (both `DOI` per spec and `doi` lowercase per
310    // real-world exports). Zotero emits uppercase; Mendeley sometimes
311    // lowercase.
312    if let Some(doi) = entry
313        .get("DOI")
314        .or_else(|| entry.get("doi"))
315        .and_then(|v| v.as_str())
316    {
317        let raw = doi.trim();
318        if !raw.is_empty() {
319            return match Ref::parse(raw) {
320                Ok(ref_) => Ok(ParsedEntry { ref_, entry_key }),
321                Err(e) => Err(ParseError::InvalidRef {
322                    raw: raw.to_string(),
323                    entry_key,
324                    source: e,
325                }),
326            };
327        }
328    }
329    // Priority 2: arXiv — `archivePrefix == "arXiv"` (CSL extension)
330    // OR the Zotero-specific `note: "arXiv:..."` shape.
331    let is_arxiv = entry
332        .get("archivePrefix")
333        .or_else(|| entry.get("archive_prefix"))
334        .and_then(|v| v.as_str())
335        .map(|s| s.eq_ignore_ascii_case("arxiv"))
336        .unwrap_or(false);
337    if is_arxiv {
338        if let Some(eprint) = entry.get("eprint").and_then(|v| v.as_str()) {
339            let raw = eprint.trim();
340            if !raw.is_empty() {
341                let with_scheme = if raw.to_ascii_lowercase().starts_with("arxiv:") {
342                    raw.to_string()
343                } else {
344                    format!("arxiv:{raw}")
345                };
346                return match Ref::parse(&with_scheme) {
347                    Ok(ref_) => Ok(ParsedEntry { ref_, entry_key }),
348                    Err(e) => Err(ParseError::InvalidRef {
349                        raw: with_scheme,
350                        entry_key,
351                        source: e,
352                    }),
353                };
354            }
355        }
356    }
357    // Fallback: scan `note` for an embedded `arXiv:NNNN.NNNNN` —
358    // Zotero often stores the arXiv id there instead of a typed
359    // field. The pattern is intentionally narrow (must follow the
360    // canonical "arXiv:" prefix); free-text DOIs in notes are NOT
361    // mined here.
362    if let Some(note) = entry.get("note").and_then(|v| v.as_str()) {
363        if let Some(idx) = note.to_ascii_lowercase().find("arxiv:") {
364            let tail = &note[idx + "arxiv:".len()..];
365            // Take chars matching the arXiv id alphabet (digits / dot /
366            // slash / letters / hyphen) — stop at the first separator
367            // so the rest of the note is ignored.
368            let id: String = tail
369                .chars()
370                .take_while(|c| matches!(c, '0'..='9' | '.' | '/' | 'a'..='z' | 'A'..='Z' | '-'))
371                .collect();
372            if !id.is_empty() {
373                let with_scheme = format!("arxiv:{id}");
374                return match Ref::parse(&with_scheme) {
375                    Ok(ref_) => Ok(ParsedEntry { ref_, entry_key }),
376                    Err(e) => Err(ParseError::InvalidRef {
377                        raw: with_scheme,
378                        entry_key,
379                        source: e,
380                    }),
381                };
382            }
383        }
384    }
385    // #500, CSL-JSON half. The BibTeX parser learned to say "this entry HAS an
386    // identifier I cannot use" and this one did not, so the same PubMed record
387    // exported as CSL-JSON still got "entry has no DOI / arXiv id" -- the claim
388    // #500 exists to stop doiget making, surviving in the format a Zotero user
389    // is most likely to hand it.
390    if let Some((kind, value)) = csl_unsupported_identifier(entry) {
391        return Err(ParseError::UnsupportedIdentifier {
392            kind,
393            value,
394            entry_key,
395        });
396    }
397    Err(ParseError::NoIdentifier { entry_key })
398}

400/// The CSL-JSON counterpart of [`unsupported_identifier`]: an identifier
401/// doiget recognises and cannot resolve yet (#500).
402///
403/// CSL-JSON has no standard PMID field, so exporters improvise. Zotero writes
404/// `PMID: 9659853` into `note`; some tools emit a top-level `PMID` key. Both
405/// are checked, and the note scan mirrors the arXiv one directly above it.
406fn csl_unsupported_identifier(entry: &serde_json::Value) -> Option<(&'static str, String)> {
407    let direct = |name: &str| -> Option<String> {
408        let v = entry.get(name)?;
409        let s = match v {
410            serde_json::Value::String(s) => s.trim().to_string(),
411            serde_json::Value::Number(n) => n.to_string(),
412            _ => return None,
413        };
414        (!s.is_empty()).then_some(s)
415    };
416    for (field, kind) in [
417        ("PMID", "PMID"),
418        ("pmid", "PMID"),
419        ("PMCID", "PMCID"),
420        ("pmcid", "PMCID"),
421    ] {
422        if let Some(v) = direct(field) {
423            return Some((kind, v));
424        }
425    }
426
427    // Zotero's `note` carries `PMID: 9659853` / `PMCID: PMC1234567`.
428    let note = entry.get("note").and_then(|v| v.as_str())?;
429    for (needle, kind) in [("pmcid:", "PMCID"), ("pmid:", "PMID")] {
430        let lower = note.to_ascii_lowercase();
431        if let Some(at) = lower.find(needle) {
432            let value: String = note[at + needle.len()..]
433                .trim_start()
434                .chars()
435                .take_while(|c| c.is_ascii_alphanumeric())
436                .collect();
437            if !value.is_empty() {
438                return Some((kind, value));
439            }
440        }
441    }
442    None
443}

445/// Parse a BibTeX / BibLaTeX document via the `biblatex` crate
446/// (ADR-0030 D2). One `@entrytype{KEY, …}` produces one entry;
447/// `entry_key` is the citation key verbatim.
448///
449/// A whole-input parse failure (malformed BibTeX the `biblatex` crate
450/// rejects) is returned as a single-element [`ParseError::Decode`] so
451/// the caller's exit-code path counts it as one error rather than
452/// zero — matching the CSL-JSON behaviour.
453///
454/// Identifier-pick priority per ADR-0030 D3: `doi` field, then
455/// `eprint` (arXiv). See `parse_bibtex_entry`.
456pub fn parse_bibtex(text: &str) -> Vec<Result<ParsedEntry, ParseError>> {
457    let bib = match Bibliography::parse(text) {
458        Ok(b) => b,
459        Err(e) => {
460            return vec![Err(ParseError::Decode {
461                format: "bibtex",
462                message: e.to_string(),
463            })]
464        }
465    };
466    bib.iter()
467        .map(|entry| parse_bibtex_entry(entry, Some(entry.key.clone())))
468        .collect()
469}

471/// Pick the highest-priority identifier on a single BibTeX entry and
472/// parse it. Honors ADR-0030 D3 priority (`doi` > arXiv `eprint`).
473fn parse_bibtex_entry(
474    entry: &biblatex::Entry,
475    entry_key: Option<String>,
476) -> Result<ParsedEntry, ParseError> {
477    // Priority 1: `doi` field. The typed accessor formats the chunk
478    // value to a `String`; `Err` means the field is absent or not a
479    // plain string, which we treat as "no DOI here" and fall through.
480    if let Ok(doi) = entry.doi() {
481        let raw = doi.trim();
482        if !raw.is_empty() {
483            return match Ref::parse(raw) {
484                Ok(ref_) => Ok(ParsedEntry { ref_, entry_key }),
485                Err(e) => Err(ParseError::InvalidRef {
486                    raw: raw.to_string(),
487                    entry_key,
488                    source: e,
489                }),
490            };
491        }
492    }
493    // Priority 2: arXiv via the `eprint` field. The BibTeX convention
494    // is `eprint = {2204.12345}` with `archivePrefix = {arXiv}` (or the
495    // BibLaTeX `eprinttype = {arxiv}`). We accept the eprint as an arXiv
496    // id when the prefix names arXiv OR is absent (the dominant
497    // single-preprint-server convention); a prefix that names something
498    // else (e.g. `eprinttype = {pubmed}`) is skipped rather than
499    // parsed incorrectly.
500    if let Ok(eprint) = entry.eprint() {
501        let raw = eprint.trim();
502        if !raw.is_empty() && arxiv_eligible(entry) {
503            let with_scheme = if raw.to_ascii_lowercase().starts_with("arxiv:") {
504                raw.to_string()
505            } else {
506                format!("arxiv:{raw}")
507            };
508            return match Ref::parse(&with_scheme) {
509                Ok(ref_) => Ok(ParsedEntry { ref_, entry_key }),
510                Err(e) => Err(ParseError::InvalidRef {
511                    raw: with_scheme,
512                    entry_key,
513                    source: e,
514                }),
515            };
516        }
517    }
518    // #500: before reporting "no identifier", check for one doiget simply
519    // does not support. Saying "no DOI / arXiv id" about an entry that
520    // carries a PMID is accurate about the parser and wrong about the entry.
521    if let Some((kind, value)) = unsupported_identifier(entry) {
522        return Err(ParseError::UnsupportedIdentifier {
523            kind,
524            value,
525            entry_key,
526        });
527    }
528    Err(ParseError::NoIdentifier { entry_key })
529}

531/// An identifier doiget recognises but cannot resolve yet (#500).
532///
533/// Only classes doiget can *name*. An entry carrying some field this does not
534/// know about still reports [`ParseError::NoIdentifier`], which stays correct
535/// for it: the point is not to guess, it is to stop saying "no identifier"
536/// about the cases where there demonstrably is one.
537///
538/// `pmid = {...}` is what PubMed's own BibTeX export writes. The BibLaTeX
539/// shape is `eprint = {...}` with `eprinttype = {pubmed}`, which
540/// [`arxiv_eligible`] already refuses -- correctly, and until now silently.
541fn unsupported_identifier(entry: &biblatex::Entry) -> Option<(&'static str, String)> {
542    let field = |name: &str| -> Option<String> {
543        let v = entry.get(name)?.format_verbatim().trim().to_string();
544        (!v.is_empty()).then_some(v)
545    };
546
547    if let Some(v) = field("pmid") {
548        return Some(("PMID", v));
549    }
550    if let Some(v) = field("pmcid") {
551        return Some(("PMCID", v));
552    }
553    let names_pubmed = entry
554        .get("archiveprefix")
555        .or_else(|| entry.get("eprinttype"))
556        .is_some_and(|c| c.format_verbatim().to_ascii_lowercase().contains("pubmed"));
557    if names_pubmed {
558        if let Some(v) = field("eprint") {
559            return Some(("PMID", v));
560        }
561    }
562    None
563}

565/// Whether an `eprint` field should be interpreted as an arXiv id.
566/// True when `archivePrefix` / `eprinttype` names arXiv (case-
567/// insensitive) or is absent; false when it explicitly names a
568/// different preprint server.
569fn arxiv_eligible(entry: &biblatex::Entry) -> bool {
570    match entry
571        .get("archiveprefix")
572        .or_else(|| entry.get("eprinttype"))
573    {
574        Some(chunks) => chunks
575            .format_verbatim()
576            .to_ascii_lowercase()
577            .contains("arxiv"),
578        None => true,
579    }
580}

582#[cfg(test)]
583#[allow(clippy::expect_used, clippy::unwrap_used, clippy::panic)]
584mod tests {
585    /// #500, CSL-JSON. The BibTeX parser learned to distinguish "has no
586    /// identifier" from "has one I cannot use"; this format did not, so the
587    /// same PubMed record exported from Zotero still got the claim #500 exists
588    /// to prevent -- in the format a Zotero user is most likely to hand over.
589    #[test]
590    fn a_csl_entry_identified_only_by_a_pmid_is_not_missing_an_identifier() {
591        let doc = r#"[{"id":"Smith2020","type":"article-journal","PMID":"9659853"}]"#;
