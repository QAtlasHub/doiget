1//! `doiget list-recent [--limit=N]` subcommand — read-only most-recent
2//! listing.
3//!
4//! Reads the configured store's `.metadata/` directory via
5//! [`Store::list_recent`] and prints one row per entry on stdout, ordered
6//! most-recent first by `[doiget].fetched_at`. Network access is never
7//! required.
8//!
9//! Output is a tab-separated table with a header line. The five columns
10//! match [`EntryInfo`](doiget_core::store::EntryInfo): `safekey`, `year`,
11//! `title`, `fetched_at`, `pdf`. Missing `year` / `fetched_at` render as
12//! `-`.
13//!
14//! `pdf` is #481. Without it this command -- the only one that answers
15//! "what do I have?" without knowing the ref in advance -- showed a
16//! metadata-only entry identically to a fetched paper. Batch fifty refs
17//! with ten blocked, come back later, and the natural conclusion is fifty
18//! papers.
19
20use std::io::Write;
21
22use anyhow::{Context, Result};
23
24use doiget_core::store::{FsStore, Store};
25
26use super::resolve_store_root;
27
28/// Format string for [`chrono::DateTime`] columns. RFC3339-shaped, UTC, no
29/// fractional seconds — matches the on-disk wire format from
30/// [`docs/STORE.md`](../../../../docs/STORE.md) §2.
31const FETCHED_AT_FMT: &str = "%Y-%m-%dT%H:%M:%SZ";
32
33/// Run the `list-recent` subcommand against the configured store.
34///
35/// Emits a tab-separated table on stdout. The column order is intentionally
36/// stable for `cut(1)` consumption; future fields will be APPENDED, not
37/// inserted.
38pub fn run(
39    limit: usize,
40    missing_pdf: bool,
41    mode: super::output::OutputMode,
42    quiet_was_explicit: bool,
43) -> Result<()> {
44    // `mode` honors ADR-0017: explicit `Quiet` suppresses the TSV table
45    // but the store read still runs (so I/O failures surface as exit 1)
46    // (#203). The non-TTY *implicit* Quiet does NOT suppress —
47    // `list-recent` is artifact-class (ADR-0017 Amendment 2 / #301): the
48    // listing IS the requested output. Json body is tracked in #204.
49    let store_root = resolve_store_root()?;
50    let store = FsStore::new(store_root)?;
51    let mut entries = store
52        .list_recent(limit)
53        .context("failed to list recent store entries")?;
54    // #481: "which of my fifty need retrying?" is the question the pdf
55    // column creates, and scanning a fifty-row table by eye is the thing a
56    // flag exists to avoid.
57    if missing_pdf {
58        entries.retain(|e| !e.has_pdf());
59    }
60
61    if mode == super::output::OutputMode::Quiet && quiet_was_explicit {
62        return Ok(());
63    }
64
65    let stdout = std::io::stdout();
66    let mut out = stdout.lock();
67    if mode == super::output::OutputMode::Json {
68        // `EntryInfo` serialises `size_bytes` directly. `has_pdf` is
69        // derived and added beside it, because a consumer should not have
70        // to know that `0` and `null` both mean "no PDF" while meaning
71        // different things about the entry.
72        let entries_json: Vec<serde_json::Value> = entries
73            .iter()
74            .map(|e| {
75                let mut v = serde_json::to_value(e)
76                    .unwrap_or_else(|_| serde_json::json!({ "safekey": e.safekey.as_str() }));
77                if let Some(o) = v.as_object_mut() {
78                    o.insert("has_pdf".into(), serde_json::json!(e.has_pdf()));
79                }
80                v
81            })
82            .collect();
83        let envelope = serde_json::json!({
84            "ok": true,
85            "count": entries_json.len(),
86            "entries": entries_json,
87        });
88        let s = serde_json::to_string_pretty(&envelope)
89            .context("failed to serialize list-recent entries to JSON")?;
90        writeln!(out, "{s}").context("failed to write list-recent JSON to stdout")?;
91        return Ok(());
92    }
93    writeln!(out, "safekey\tyear\ttitle\tfetched_at\tpdf")
94        .context("failed to write list-recent header to stdout")?;
95    for e in &entries {
96        let year = e.year.map(|y| y.to_string()).unwrap_or_else(|| "-".into());
97        let fetched = e
98            .fetched_at
99            .map(|t| t.format(FETCHED_AT_FMT).to_string())
100            .unwrap_or_else(|| "-".into());
101        writeln!(
102            out,
103            "{}\t{}\t{}\t{}\t{}",
104            e.safekey.as_str(),
105            year,
106            e.title,
107            fetched,
108            pdf_cell(e)
109        )
110        .context("failed to write list-recent row to stdout")?;
111    }
112    Ok(())
113}
114
115/// Render the `pdf` column: the stored PDF size, or a dash for a
116/// metadata-only entry.
117///
118/// #481. APPENDED rather than inserted before `title`, because this
119/// module's contract is explicit that column order is stable for `cut(1)`
120/// and new fields go on the end. The issue suggested inserting it; that
121/// would break every existing `cut -f3` in the wild.
122///
123/// The size rather than a bare boolean: "1.0 MB" and "-" answer "did I get
124/// it?" equally well, and the size additionally answers "is this the paper
125/// or a one-page stub?", which is the next question and costs nothing.
126pub(crate) fn pdf_cell(e: &doiget_core::store::EntryInfo) -> String {
127    match e.size_bytes {
128        // `None` is "no `[doiget]` table", a different unknown from
129        // "0 bytes stored". Both mean no PDF; only one means the entry
130        // lacks the table that would say so.
131        None => "?".to_string(),
132        Some(0) => "-".to_string(),
133        #[allow(clippy::cast_precision_loss)]
134        Some(n) if n >= 1_048_576 => format!("{:.1} MB", n as f64 / 1_048_576.0),
135        #[allow(clippy::cast_precision_loss)]
136        Some(n) if n >= 1024 => format!("{:.1} kB", n as f64 / 1024.0),
137        Some(n) => format!("{n} B"),
138    }
139}
