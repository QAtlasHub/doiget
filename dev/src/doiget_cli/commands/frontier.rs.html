1//! `doiget frontier <doi>` — gap-spotting frontier view (#295).
2//!
3//! Surfaces papers that **cite the seed DOI**, ranked by age-normalized
4//! impact (`fwci` descending). Papers already present in the local store
5//! are filtered out by default so only unread candidates are shown.
6//!
7//! ## Output
8//!
9//! - **Human mode**: tab-separated table `fwci | year | oa | doi | title`.
10//! - **JSON mode**: `{ seed_doi, seed_title, seed_openalex_id, total_citing,
11//!   count, results: [PaperHit...] }`.
12//!
13//! ## Store exclusion
14//!
15//! Each candidate DOI is mapped to its safekey and checked against
16//! `<store_root>/<safekey>.pdf`. Matches are silently dropped (the purpose
17//! of frontier is to surface the *unread* portion of the neighbourhood).
18

22use doiget_core::discovery::{frontier_view, FrontierQuery, FrontierResults};

29/// OpenAlex API base; overridable via `DOIGET_OPENALEX_BASE` (tests).
30const OPENALEX_DEFAULT_BASE: &str = "https://api.openalex.org";
31
32/// Entry point for `doiget frontier <doi>`.
33#[allow(clippy::print_stdout, clippy::print_stderr)]
34pub async fn run(
35    doi_str: String,
36    limit: usize,
37    from_year: Option<i32>,
38    mode: OutputMode,
39    quiet_was_explicit: bool,
40) -> Result<()> {
41    // #492 / ADR-0049: previously a bare `anyhow!`, which exited 1 with no
42    // `error[INVALID_REF]:` line. Routed through the shared parser so the
43    // message and the exit code match every other ref-taking command.
44    let seed_doi = match super::parse_ref_or_exit(&doi_str)? {
45        doiget_core::Ref::Doi(d) => d,
46        doiget_core::Ref::Arxiv(_) => {
47            // Same shape as `graph`: OpenAlex keys the citation graph by
48            // DOI, so an arXiv id is argument misuse rather than a parse
49            // failure — `docs/ERRORS.md` §4 exit 2.
50            super::output::print_err(format_args!(
51                "error: doiget frontier requires a DOI seed; arXiv ids are not in \
52                 OpenAlex's referenced_works keyspace"
53            ));
54            return Err(anyhow::Error::new(CliExit(cli_exit_code(
55                doiget_core::ErrorCode::InvalidRef,
56            ))));
57        }
58    };
59
60    let base = {
61        let raw = std::env::var("DOIGET_OPENALEX_BASE")
62            .unwrap_or_else(|_| OPENALEX_DEFAULT_BASE.to_string());
63        url::Url::parse(&raw)
64            .with_context(|| format!("DOIGET_OPENALEX_BASE is not a URL: {raw}"))?
65    };
66    // `unwrap_or_default`, not the placeholder: an unset address means
67    // `mailto` is omitted rather than sent as `doiget@localhost`. Routed
68    // through the core resolver so config.toml's rung counts (#504).
69    let contact_email = doiget_core::orchestrator::configured_contact_email().unwrap_or_default();
70
71    let harness = FetchHarness::from_env().context("building fetch harness")?;
72    harness
73        .log_session_start(Some(&doi_str))
74        .context("logging session start")?;
75    let ctx = harness.fetch_context();
76
77    let query = FrontierQuery {
78        seed_doi: seed_doi.clone(),
79        limit: limit.clamp(1, 200),
80        min_year: from_year,
81    };
82
83    let outcome = frontier_view(&query, &base, &contact_email, &ctx).await;
84    harness.log_session_end(
85        outcome.is_ok(),
86        Some(&doi_str),
87        outcome.as_ref().err().map(|e| ErrorCode::from(e).as_wire()),
88    );
89
90    let mut results = match outcome {
91        Ok(r) => r,
92        Err(e) => {
93            let code = ErrorCode::from(&e);
94            eprintln!("error[{}]: {e}", code.as_wire());
95            return Err(anyhow::Error::new(CliExit(cli_exit_code(code))));
96        }
97    };
98
99    // Filter out papers already in the local store.
100    match resolve_store_root() {
101        Ok(store_root) => {
102            results.hits.retain(|hit| {
103                let Some(ref doi_s) = hit.doi else {
104                    return true;
105                };
106                let Ok(d) = doiget_core::Doi::parse(doi_s) else {
107                    return true;
108                };
109                let safekey = doiget_core::Ref::Doi(d).safekey();
110                !store_root
111                    .join(format!("{}.pdf", safekey.as_str()))
112                    .exists()
113            });
114        }
115        Err(e) => {
116            // Don't silently present already-fetched papers as "frontier":
117            // tell the user the store-exclusion filter was skipped (review #352).
118            eprintln!(
119                "warning: could not resolve the local store root ({e}); \
120                 frontier results are NOT filtered against already-fetched papers"
121            );
122        }
123    }
124
125    if mode == OutputMode::Quiet && quiet_was_explicit {
126        return Ok(());
127    }
128
129    let stdout = std::io::stdout();
130    let mut out = stdout.lock();
131
132    if mode == OutputMode::Json {
133        let envelope = json_envelope(&doi_str, &results);
134        let s = serde_json::to_string(&envelope).context("serialise frontier results to JSON")?;
135        writeln!(out, "{s}").context("write frontier JSON to stdout")?;
136        return Ok(());
137    }
138
139    // Human table: fwci first (the age-normalized signal that matters most),
140    // then year / oa / doi / title.
141    writeln!(out, "fwci\tyear\toa\tdoi\ttitle").context("write frontier header")?;
142    for hit in &results.hits {
143        let fwci = hit
144            .fwci
145            .map(|f| format!("{f:.2}"))
146            .unwrap_or_else(|| "-".into());
147        let year = hit
148            .year
149            .map(|y| y.to_string())
150            .unwrap_or_else(|| "-".into());
151        let oa = hit.oa_status.as_deref().unwrap_or("-");
152        let doi = hit.doi.as_deref().unwrap_or("-");
153        writeln!(out, "{fwci}\t{year}\t{oa}\t{doi}\t{}", hit.title)
154            .context("write frontier row")?;
155    }
156    Ok(())
157}
158
159fn json_envelope(seed_doi: &str, results: &FrontierResults) -> serde_json::Value {
160    serde_json::json!({
161        "seed_doi": seed_doi,
162        "seed_title": results.seed_title,
163        "seed_openalex_id": results.seed_openalex_id,
164        "total_citing": results.total_citing,
165        "count": results.hits.len(),
166        "results": results.hits,
167    })
168}
