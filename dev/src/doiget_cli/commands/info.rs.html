1//! `doiget info <ref>` subcommand — read-only metadata inspection.
2//!
3//! Reads the metadata TOML for a given [`Ref`](doiget_core::Ref) from the
4//! [`FsStore`] and
5//! prints it on stdout. Network access is never required.
6//!
7//! Per [`docs/PUBLIC_API.md`](../../../../docs/PUBLIC_API.md) §2 the read goes
8//! through the [`Store::read`] trait method, so any future store backend
9//! (e.g. a SQLite index in Phase 2) is a drop-in replacement.
10
11use std::io::Write;
12
13use anyhow::{bail, Context, Result};
14
15use doiget_core::store::{FsStore, Store};
16
17use super::resolve_store_root;
18
19/// Run the `info` subcommand against the configured store.
20///
21/// `input` is the user-supplied ref string (e.g. `"10.1234/example"`,
22/// `"arxiv:2401.12345"`, or any of the schemes accepted by
23/// [`Ref::parse`](doiget_core::Ref::parse)).
24///
25/// On success, the entry's [`Metadata`](doiget_core::store::Metadata) is
26/// written to stdout as TOML. On a missing entry, the function returns an
27/// error so the CLI exits non-zero — the caller (a shell pipeline) can
28/// distinguish "entry not in store" from "entry was empty".
29pub fn run(input: String, mode: super::output::OutputMode, quiet_was_explicit: bool) -> Result<()> {
30    // `mode` honors ADR-0017: explicit `Quiet` skips emission but still
31    // resolves the entry so the "not-found → exit non-zero" contract
32    // continues to hold (#203). The non-TTY *implicit* Quiet does NOT
33    // suppress — `info` is artifact-class (ADR-0017 Amendment 2 / #301):
34    // its stdout IS the requested metadata, which an agent / pipe / ssh
35    // caller needs. `Json` serialises `Metadata` directly (it carries
36    // `Serialize`); the wire form is the field-name JSON of the on-disk
37    // TOML (#204).
38    let ref_ = super::parse_ref_or_exit(&input)?;
39    let safekey = ref_.safekey();
40
41    let store_root = resolve_store_root()?;
42    let store = FsStore::new(store_root)?;
43
44    let metadata = store
45        .read(&safekey)
46        .with_context(|| format!("failed to read store entry for {input}"))?;
47
48    match metadata {
49        Some(m) => {
50            if mode == super::output::OutputMode::Quiet && quiet_was_explicit {
51                // Explicit Quiet only (--quiet / -q / --mode quiet /
52                // DOIGET_MODE=quiet): existence-check, exit 0, no stdout.
53                return Ok(());
54            }
55            // Workspace lints deny `print_stdout` (the `print!`/`println!`
56            // macros) so JSON-RPC frames never collide with diagnostics.
57            // `writeln!` / `write!` against an explicit `stdout().lock()`
58            // is the sanctioned escape hatch — the caller chose stdout
59            // explicitly. See `docs/SECURITY.md` §3 / ADR-0001.
60            let stdout = std::io::stdout();
61            let mut out = stdout.lock();
62            if mode == super::output::OutputMode::Json {
63                let envelope = serde_json::json!({
64                    "ok": true,
65                    "ref": input,
66                    "safekey": safekey.as_str(),
67                    "metadata": m,
68                });
69                let s = serde_json::to_string_pretty(&envelope)
70                    .context("failed to serialize metadata to JSON for stdout")?;
71                writeln!(out, "{s}").context("failed to write metadata JSON to stdout")?;
72                return Ok(());
73            }
74            // Human (default): re-serialize to TOML for stdout. We use
75            // `toml::to_string_pretty` for human readability; this is NOT
76            // the §7-normalized form written to disk, but `info` is a
77            // presentation surface, not a round-tripper.
78            let s = toml::to_string_pretty(&m)
79                .context("failed to serialize metadata to TOML for stdout")?;
80            write!(out, "{s}").context("failed to write metadata to stdout")?;
81            // toml::to_string_pretty already emits a trailing newline, but
82            // be defensive in case a future toml-rs revision changes that.
83            if !s.ends_with('\n') {
84                writeln!(out).context("failed to write trailing newline")?;
85            }
86            Ok(())
87        }
88        None => bail!("no entry for {input}"),
89    }
90}
