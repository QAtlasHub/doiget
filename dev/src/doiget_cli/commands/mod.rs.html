1//! Subcommand implementations for the `doiget` CLI.
2//!
3//! Each module corresponds to a single `clap` subcommand declared in
4//! `main.rs`. The dispatch table in `main.rs` calls `run(...)` on the
5//! matching module. Subcommands return `anyhow::Result<()>`; any error
6//! surfaces via the CLI's top-level error reporter (stderr).
7//!
8//! ## Phase 1 surface (so far)
9//!
10//! - [`audit_log`] — `doiget audit-log --verify` recomputes the SHA-256 hash
11//!   chain on the provenance log and reports any mismatches.
12//! - [`batch`] — `doiget batch <path>` multi-ref orchestrator (rate-bounded).
13//! - [`bib`] — `doiget bib <ref>` BibTeX exporter (Phase 2 starter).
14//! - [`cite`] — `doiget cite <ref>` live-resolve BibTeX (doi2bib-style).
15//! - [`config`] — `doiget config show/path/doctor`.
16//! - [`csl`] — `doiget csl <ref>` exports a stored entry as CSL JSON 1.0.
17//! - [`fetch`] — `doiget fetch <ref>` orchestrator (arXiv E2E + DOI metadata-only).
18//! - [`info`] — prints a stored entry's `Metadata` as TOML on stdout.
19//! - [`list_recent`] — prints up to N most-recently-fetched entries.
20//! - [`search`] — case-insensitive substring search over stored metadata.
21//!
22//! `serve` has no module here: it delegates straight to `doiget-mcp`. It
23//! is wired in `main.rs` and is what every `docs/INTEGRATION/` guide tells
24//! users to run.
25
26/// Parse a ref, or render the failure in the `docs/ERRORS.md` §3
27/// "Researcher (CLI human)" form and return the CLI exit error.
28///
29/// #477. `error[CODE]: message` held on `fetch` alone -- #119 did it there
30/// and nowhere else -- while `info`, `link`, `cite`, `text`, `tag`, `bib`,
31/// `csl` and `source` each wrote their own `with_context("invalid ref: …")`
32/// and let `anyhow` print a bare `Error:` plus a `Caused by:` chain. The
33/// closed error-code set is a load-bearing promise of this project: a
34/// caller is told it can key off `error[CODE]:`, and on eight of nine
35/// commands there was no code to key off, while the `Caused by:` chain
36/// leaked internal error types that are in no contract.
37///
38/// Same shape as `render_fetch_error` and for the same reason -- one
39/// renderer, so a future change to the contract cannot reach some call
40/// sites and miss others.
41///
42/// # Errors
43///
44/// Always, when parsing fails: an [`anyhow::Error`] wrapping
45/// [`CliExit`](fetch::CliExit) with the `INVALID_REF` exit code. The
46/// message has already been written to stderr.
47pub fn parse_ref_or_exit(input: &str) -> anyhow::Result<doiget_core::Ref> {
48    match doiget_core::Ref::parse(input) {
49        Ok(r) => Ok(r),
50        // #500: a PubMed id is not a malformed DOI. Say which commands take
51        // one, rather than calling it invalid.
52        Err(_) if doiget_core::pubmed::PubmedId::parse(input).is_some() => {
53            output::print_err(format_args!(
54                "error[INVALID_REF]: {input} is a PubMed id; `doiget fetch` and `doiget cite` \
55                 resolve it to its DOI -- this command takes the DOI itself"
56            ));
57            Err(anyhow::Error::new(fetch::CliExit(fetch::cli_exit_code(
58                doiget_core::ErrorCode::InvalidRef,
59            ))))
60        }
61        Err(e) => {
62            render_ref_parse_error(&e);
63            Err(anyhow::Error::new(fetch::CliExit(fetch::cli_exit_code(
64                doiget_core::ErrorCode::InvalidRef,
65            ))))
66        }
67    }
68}

70/// [`parse_ref_or_exit`] that also takes a PubMed id (#500, ADR-0061),
71/// turning it into the DOI PubMed lists for it with one E-utilities lookup.
72/// `network: false` (a dry run) refuses the lookup and says so, rather than
73/// making a request the caller was promised would not happen.
74pub async fn parse_ref_or_pubmed(input: &str, network: bool) -> anyhow::Result<doiget_core::Ref> {
75    use doiget_core::pubmed::{Lookup, PubmedId};
76    let Some(id) = PubmedId::parse(input) else {
77        return parse_ref_or_exit(input);
78    };
79    let fail = |code: doiget_core::ErrorCode, why: &str| {
80        output::print_err(format_args!("error[{}]: {why}", code.as_wire()));
81        Err(anyhow::Error::new(fetch::CliExit(fetch::cli_exit_code(
82            code,
83        ))))
84    };
85    if !network {
86        return fail(
87            doiget_core::ErrorCode::InvalidRef,
88            &format!(
89                "{} becomes a DOI through one NCBI lookup, and --dry-run makes no request; pass the DOI",
90                id.display()
91            ),
92        );
93    }
94    let ctx = fetch::build_resolve_context()?;
95    match doiget_core::pubmed::lookup_in_session(&id, &ctx).await {
96        Ok(Lookup::Doi(doi)) => {
97            output::print_err(format_args!(
98                "note: {} is DOI {}",
99                id.display(),
100                doi.as_str()
101            ));
102            Ok(doiget_core::Ref::Doi(doi))
103        }
104        Ok(found @ Lookup::NoRecord) => fail(
105            doiget_core::ErrorCode::NotFound,
106            &found.reason(&id).unwrap_or_default(),
107        ),
108        Ok(found) => fail(
109            doiget_core::ErrorCode::NotImplemented,
110            &found.reason(&id).unwrap_or_default(),
111        ),
112        Err(e) => {
113            let code = doiget_core::ErrorCode::from(&e);
114            fail(
115                code,
116                &format!("looking up {} at NCBI failed: {e}", id.display()),
117            )
118        }
119    }
120}

122/// The renderer on its own, for call sites holding a
123/// [`doiget_core::RefParseError`] that did not come from
124/// [`parse_ref_or_exit`].
125///
126/// Both now take the exit code from `fetch::cli_exit_code(InvalidRef)`,
127/// which is **2** since #492 / ADR-0049. Before that `fetch` fell to the
128/// generic `_ => 1` arm while `graph` hard-coded the 2 that
129/// `docs/ERRORS.md` §4 prescribes, so one binary gave two answers for the
130/// same input and each site's comment claimed agreement with the other.
131///
132/// One renderer and one exit code. `every_ref_taking_command_exits_2_for_
133/// an_invalid_ref` is what keeps it that way.
134pub fn render_ref_parse_error(e: &doiget_core::RefParseError) {
135    output::print_err(format_args!(
136        "error[{}]: invalid ref: {e}",
137        doiget_core::ErrorCode::InvalidRef.as_wire()
138    ));
139}

176/// Resolve the path of the user `config.toml`, for messages that need to
177/// name the file the user must edit.
178///
179/// Same resolution as [`config::ResolvedConfig::from_env`]
180/// (`<dirs::config_dir()>/doiget/config.toml`), kept here so the fetch-side
181/// denial help (issue #405) and `config doctor` cannot drift. Returns
182/// `None` only when the platform has no config dir at all, in which case
183/// callers should fall back to naming the file generically — a missing
184/// config dir must never turn an advisory line into a hard error.
185pub(crate) fn user_config_path() -> Option<Utf8PathBuf> {
186    // MUST stay `config_dir_utf8` — the resolver the READER uses. This
187    // shipped as `dirs::config_dir()`, which ignores `XDG_CONFIG_HOME` on
188    // Windows, so the denial help named `%APPDATA%\doiget\config.toml`
189    // while `build_http_client` was loading the XDG one. Naming the wrong
190    // file is worse than naming none, and it is the whole point of the
191    // #405 help line. Same fix as `ResolvedConfig::from_env`.
192    Some(
193        fetch::config_dir_utf8()
194            .ok()?
195            .join("doiget")
196            .join("config.toml"),
197    )
198}

200/// Where a resolved store root came from (#441).
201///
202/// Reported by `doiget config doctor` so that a setting which did nothing
203/// can no longer look like a setting that worked. That was the sharpest
204/// part of #441: `config init` recommended `[store] root`, `doctor`
205/// confirmed the recommendation, and the value was never read.
206#[derive(Debug, Clone, Copy, PartialEq, Eq)]
207pub(crate) enum StoreRootSource {
208    /// `DOIGET_STORE_ROOT` (also how `--store-root` is applied).
209    Env,
210    /// `[store] root` in the user's `config.toml`.
211    ConfigFile,
212    /// Nothing set it — `./papers` under the cwd (ADR-0036).
213    CwdDefault,
214}
215
216impl StoreRootSource {
217    /// Short label for `config show` / `doctor`.
218    pub(crate) fn label(self) -> &'static str {
219        match self {
220            Self::Env => "DOIGET_STORE_ROOT",
221            Self::ConfigFile => "[store] root in config.toml",
222            Self::CwdDefault => "default: ./papers under the cwd",
223        }
224    }
225}

227/// Resolve the on-disk store root.
228///
229/// Resolution order (ADR-0036, `docs/CONFIG.md` §4):
230///
231/// 1. `DOIGET_STORE_ROOT`, if set and non-empty. `--store-root` is applied
232///    by writing this variable, so the flag rides the same rung.
233/// 2. `[store] root` in the user's `config.toml`, expanded via
234///    [`doiget_core::user_extension::expand_store_root`].
235/// 3. `./papers` — `papers/` directly under the current working directory
236///    (#344 / ADR-0036), so fetched artifacts are visible where the user
237///    (or an LLM agent) is working rather than hidden in a far-off home
238///    directory.
239///
240/// Rung 2 is new in #441. It was documented from the start — ADR-0036
241/// states the order, `docs/CONFIG.md` §3 lists the key, `config init`
242/// writes it into the template it generates and `config doctor` recommends
243/// it — and read by nothing, so the store silently kept following the cwd.
244/// The old doc comment on this function said the config rung "lands with
245/// the `config` subcommand"; that subcommand shipped in 0.8.8 without it.
246pub(crate) fn resolve_store_root() -> Result<Utf8PathBuf> {
247    resolve_store_root_with_source().map(|(root, _)| root)
248}
249
250/// [`resolve_store_root`] plus which rung answered.
251pub(crate) fn resolve_store_root_with_source() -> Result<(Utf8PathBuf, StoreRootSource)> {
252    if let Ok(s) = std::env::var("DOIGET_STORE_ROOT") {
253        if let Some(root) = usable_store_root_env(&s) {
254            return Ok((Utf8PathBuf::from(root), StoreRootSource::Env));
255        }
256    }
257    if let Some(root) = store_root_from_config() {
258        return Ok((root, StoreRootSource::ConfigFile));
259    }
260    let cwd = std::env::current_dir().context(
261        "could not determine the current working directory for the default store root (set \
262            DOIGET_STORE_ROOT to choose an explicit store location)",
263    )?;
264    Utf8PathBuf::from_path_buf(cwd)
265        .map(|d| (d.join("papers"), StoreRootSource::CwdDefault))
266        .map_err(|p| anyhow::anyhow!("current directory path is not UTF-8: {}", p.display()))
267}

269/// The path a `DOIGET_STORE_ROOT` value names, or `None` when the value must
270/// be treated as unset: empty, whitespace, or an unexpanded `${...}`
271/// placeholder. A Desktop-Extension config left blank can pass the literal
272/// `${user_config.store_root}` (#369), and `export DOIGET_STORE_ROOT=${X:-}`
273/// in a script exports an empty string (#613); neither may become a path.
274fn usable_store_root_env(raw: &str) -> Option<&str> {
275    let s = raw.trim();
276    (!s.is_empty() && !s.contains("${")).then_some(s)
277}

279/// `DOIGET_STORE_ROOT` when it is set but ignored by
280/// [`resolve_store_root_with_source`], so `config doctor` can say so instead
281/// of reporting the fallback as though nothing had been set (#613).
282pub(crate) fn ignored_store_root_env() -> Option<String> {
283    std::env::var("DOIGET_STORE_ROOT")
284        .ok()
285        .filter(|raw| usable_store_root_env(raw).is_none())
286}

288/// `[store] root` from the user's `config.toml`, if any.
289///
290/// Does not fail the command on a malformed file — the store root is
291/// resolved by every subcommand, and one bad line should not be a total
292/// outage — but it does NOT stay quiet about it.
293///
294/// The previous comment here justified silence by claiming "a parse error
295/// surfaces with a proper diagnostic on the network path that owns this
296/// file". The #468 review checked that claim and it is false for most
297/// callers: `list-recent`, `search`, `info`, `tag`, `bib` and `csl` never
298/// build an HTTP client, so that diagnostic never runs for them. TOML fails
299/// the whole document, so a typo anywhere — even under `[network]` — makes
300/// `[store] root` unreadable, and the command silently used `./papers`
301/// under the cwd as though nothing had been configured.
302///
303/// `search` then reports zero results, indistinguishable from an empty
304/// library. Worse, `tag` **writes** metadata into the wrong store and
305/// reports success. A warning is the minimum; the resolved source is also
306/// reported by `config doctor`.
307fn store_root_from_config() -> Option<Utf8PathBuf> {
308    let path = user_config_path()?;
309    let cfg = match doiget_core::user_extension::load(&path) {
310        Ok(c) => c,
311        Err(e) => {
312            tracing::warn!(
313                path = %path,
314                error = %e,
315                "config.toml could not be read; [store] root ignored and the default store root used instead. Run `doiget config doctor` to see which root is in effect."
316            );
317            return None;
318        }
319    };
320    let raw = cfg.store_root?;
321    Some(doiget_core::user_extension::expand_store_root(&raw))
322}

324/// How `cite` / `bib` key an entry and which extra fields they add (#610).
325///
326/// Every field is opt-in; the default is the safekey and no `file`, as
327/// before. Unset options fall back to `[cite] key_template` /
328/// `[cite] file_field` in `config.toml`.
329#[derive(Debug, Clone, Default)]
330pub struct KeyOptions {
331    /// `--key`: an explicit key, single-ref only.
332    pub key: Option<String>,
333    /// `--key-template`, e.g. `{author}{year}{title_word}`.
334    pub template: Option<String>,
335    /// `--file-field`, e.g. `refs/{key}.pdf`: a `file = {...}` field.
336    pub file_field: Option<String>,
337    /// `--file-field-always`: add `file` even when the path does not exist.
338    pub file_field_always: bool,
339    /// `--journal-abbrev iso4`: add biblatex `shortjournal` from the
340    /// record's own abbreviation (#611). `journal` keeps the full title.
341    pub journal_abbrev: bool,
342    /// Entries that got no `shortjournal`, reported once by
343    /// [`KeyOptions::report`].
344    pub missing_abbrev: MissingAbbrev,
345}
346
347impl KeyOptions {
348    /// Fill unset options from `[cite]` in `config.toml` and validate the
349    /// template, so a typo fails before any network work.
350    ///
351    /// # Errors
352    ///
353    /// A template with an unknown placeholder or an unclosed `{`, naming
354    /// where it came from.
355    pub fn with_config_defaults(mut self) -> Result<Self> {
356        if let Some(k) = &self.key {
357            if !doiget_core::store::citekey::is_valid_key(k) {
358                anyhow::bail!(
359                    "--key {k:?} cannot be a BibTeX key: use only letters, digits and - _ : . + /"
360                );
361            }
362        }
363        if self.template.is_none() || self.file_field.is_none() {
364            if let Some(path) = user_config_path() {
365                // A config that does not parse must not make `[cite]` vanish
366                // silently -- TOML fails the whole document, so a typo under
367                // `[network]` would otherwise drop the key template with no
368                // word said (the #468 lesson, review of #622).
369                match doiget_core::user_extension::load(&path) {
370                    Ok(cfg) => {
371                        if self.template.is_none() && self.key.is_none() {
372                            self.template = cfg.cite_key_template;
373                        }
374                        if self.file_field.is_none() {
375                            self.file_field = cfg.cite_file_field;
376                        }
377                    }
378                    Err(e) => output::print_err(format_args!(
379                        "warning: {path} could not be read ({e}); its [cite] defaults are not \
380                         applied. Run `doiget config doctor`."
381                    )),
382                }
383            }
384        }
385        if let Some(t) = &self.template {
386            doiget_core::store::citekey::validate_template(t).with_context(|| {
387                format!("key template {t:?} (--key-template or [cite] key_template)")
388            })?;
389        }
390        Ok(self)
391    }
392
393    /// Report, once, what the render could not do (entries without an
394    /// abbreviation on record). Call after the last entry.
395    pub fn report(&self) {
396        report_missing_abbrev(&self.missing_abbrev);
397    }
398
399    /// Render `m` as BibTeX under the chosen key, recording it in `used` and
400    /// suffixing it `a`, `b`, ... on a collision.
401    ///
402    /// # Errors
403    ///
404    /// Only an invalid template, which [`KeyOptions::with_config_defaults`]
405    /// already refused.
406    pub fn bibtex(
407        &self,
408        m: &doiget_core::store::Metadata,
409        safekey: &str,
410        used: &mut std::collections::HashSet<String>,
411    ) -> Result<String> {
412        use doiget_core::store::citekey;
413        let key = match (&self.key, &self.template) {
414            (Some(k), _) => k.clone(),
415            (None, Some(t)) => citekey::render_key(t, m, safekey)?,
416            (None, None) => safekey.to_string(),
417        };
418        let key = citekey::disambiguate(key, used);
419        let file = self.file_field.as_ref().and_then(|pattern| {
420            let path = pattern.replace("{key}", &key).replace("{safekey}", safekey);
421            (self.file_field_always || camino::Utf8Path::new(&path).exists()).then_some(path)
422        });
423        let short = if self.journal_abbrev {
424            short_journal(m, &key, &self.missing_abbrev)
425        } else {
426            None
427        };
428        let extra: Vec<(&str, &str)> = short
429            .iter()
430            .map(|s| ("shortjournal", s.as_str()))
431            .chain(file.iter().map(|f| ("file", f.as_str())))
432            .collect();
433        Ok(doiget_core::store::render::to_bibtex_with_fields(
434            &key, m, &extra,
435        ))
436    }
437}

439/// Entries rendered with `--journal-abbrev` whose record carries no
440/// abbreviation: `(key, venue, source)`. Collected so a multi-entry export
441/// reports them once rather than one line per entry (review of #623).
442pub type MissingAbbrev = std::cell::RefCell<Vec<(String, String, String)>>;
443
444/// The ISO 4 abbreviation for `m`'s venue as its record gave it, or `None`
445/// -- recorded in `missing` when the entry has a venue -- never a guessed
446/// abbreviation (#611).
447pub(crate) fn short_journal(
448    m: &doiget_core::store::Metadata,
449    key: &str,
450    missing: &MissingAbbrev,
451) -> Option<String> {
452    let short = m.doiget.as_ref().and_then(|d| d.short_venue.clone());
453    if short.is_none() {
454        if let Some(venue) = m.venue.as_deref().filter(|v| !v.is_empty()) {
455            let source = m
456                .doiget
457                .as_ref()
458                .map_or("its", |d| d.source.as_str())
459                .to_string();
460            missing
461                .borrow_mut()
462                .push((key.to_string(), venue.to_string(), source));
463        }
464    }
465    short
466}

468/// Say, once, which entries got no `shortjournal` and why.
469pub(crate) fn report_missing_abbrev(missing: &MissingAbbrev) {
470    let list = missing.borrow();
471    let why = "an abbreviation is only what the record itself carries (Crossref \
472               short-container-title), never a guess; an entry stored before 0.9 has none \
473               recorded until it is re-fetched";
474    match list.as_slice() {
475        [] => {}
476        [(key, venue, source)] => output::print_err(format_args!(
477            "note: {key}: no abbreviation for {venue:?} in its {source} record, so no \
478             shortjournal; {why}"
479        )),
480        many => {
481            let shown: Vec<String> = many
482                .iter()
483                .take(5)
484                .map(|(k, v, _)| format!("{k} ({v:?})"))
485                .collect();
486            let more = many.len().saturating_sub(shown.len());
487            output::print_err(format_args!(
488                "note: {} entries have no abbreviation on record, so no shortjournal: {}{}; {why}",
489                many.len(),
490                shown.join(", "),
491                if more > 0 {
492                    format!(", and {more} more")
493                } else {
494                    String::new()
495                }
496            ));
497        }
498    }
499}

501/// The stderr lines for a #608 metadata-quality result: one `note:` per
502/// field repaired from another source, and one `warning:` naming the fields
503/// that still carry a U+FFFD. `repair_enabled` says whether any repair
504/// source was enabled, so the warning either explains that none matched or
505/// names the switch that would let doiget try.
506pub(crate) fn metadata_quality_lines(
507    repaired: &std::collections::BTreeMap<String, String>,
508    remaining: &[String],
509    repair_enabled: bool,
510) -> Vec<String> {
511    let mut lines: Vec<String> = repaired
512        .iter()
513        .map(|(field, source)| {
514            format!(
515                "note: repaired {field} from {source}: the Crossref record carries U+FFFD where \
516                 the publisher's deposit lost a character"
517            )
518        })
519        .collect();
520    let fields: Vec<&str> = remaining
521        .iter()
522        .filter_map(|f| f.strip_prefix("replacement_char:"))
523        .collect();
524    if !fields.is_empty() {
525        let hint = if repair_enabled {
526            "no enabled source had a copy that matches it character for character"
527        } else {
528            "set DOIGET_ENABLE_S2=1 to try a repair from Semantic Scholar"
529        };
530        lines.push(format!(
531            "warning: {} still carr{} U+FFFD where the publisher's deposit lost a character; {hint}",
532            fields.join(", "),
533            if fields.len() == 1 { "ies" } else { "y" },
534        ));
535    }
536    lines
537}

539/// The publisher's page for a work: Crossref's `resource.primary.URL`, else
540/// the DOI resolver (ADR-0053: a DOI link is an address, not a fetch).
541/// Shared by `missing` and `coverage`.
542pub(super) fn landing_url(ref_: &doiget_core::Ref, crossref: &serde_json::Value) -> Option<String> {
543    match ref_ {
544        doiget_core::Ref::Doi(doi) => Some(
545            crossref
546                .pointer("/resource/primary/URL")
547                .and_then(serde_json::Value::as_str)
548                .map(str::to_string)
549                .unwrap_or_else(|| format!("https://doi.org/{}", doi.as_str())),
550        ),
551        doiget_core::Ref::Arxiv(id) => Some(format!("https://arxiv.org/abs/{}", id.as_str())),
552    }
553}

564    #[test]
565    fn async_command_files_call_the_store_only_through_blocking_section() {
566        let dir = std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("src/commands");
567        let mut offenders = Vec::new();
568        let mut scanned = 0;
569        for entry in std::fs::read_dir(&dir).expect("commands dir") {
570            let path = entry.expect("entry").path();
571            if path.extension().and_then(|e| e.to_str()) != Some("rs") {
572                continue;
573            }
574            let src = std::fs::read_to_string(&path).expect("read");
575            let body = src.split("\nmod tests {").next().unwrap_or(&src);
576            if !body.contains("async fn") {
577                continue;
578            }
579            scanned += 1;
580            let flat: String = body.chars().filter(|c| !c.is_whitespace()).collect();
581            for m in [
582                "read",
583                "write",
584                "write_user_authored",
585                "list_recent",
586                "search",
587                "search_by_tag",
588            ] {
589                let calls = flat.match_indices(&*format!("store.{m}(")).count();
590                let wrapped = flat
591                    .match_indices(&*format!("blocking_section(||store.{m}("))
592                    .count()
593                    + flat
594                        .match_indices(&*format!("blocking_section(||{{store.{m}("))
595                        .count();
596                if calls != wrapped {
597                    offenders.push(format!("{}: store.{m}", path.display()));
598                }
599            }
600        }
601        assert!(scanned >= 10, "scanned only {scanned} async command files");
602        assert!(
603            offenders.is_empty(),
604            "store calls outside blocking_section: {offenders:#?}"
605        );
606    }
