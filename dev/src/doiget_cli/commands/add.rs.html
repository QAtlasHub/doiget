1//! `doiget add` -- put a hand-downloaded PDF in the store (#606).
2//!
3//! `doiget add <ref> <file.pdf>` adds one file; `doiget add --from-dir DIR`
4//! matches a downloads folder against the entries still missing a PDF --
5//! those of `--refs FILE`, else every store entry without one -- by the id in
6//! each file's name (`BF01340294.pdf` for `10.1007/BF01340294`). The folder
7//! form only prints its plan unless `--apply` is given.
8//!
9//! Every check and the recorded provenance are `doiget_core::user_pdf`'s:
10//! the PDF is never read beyond its `%PDF-` magic (ADR-0003), and the entry
11//! says `origin = "user-supplied"`, `license = "unknown"`.
12

24/// Entry point for `doiget add`.
25///
26/// # Errors
27///
28/// Misuse (exit 2) for a bad argument combination; otherwise the number of
29/// files that could not be added.
30pub async fn run(
31    ref_: Option<String>,
32    file: Option<Utf8PathBuf>,
33    from_dir: Option<Utf8PathBuf>,
34    refs: Option<Utf8PathBuf>,
35    force: bool,
36    apply: bool,
37) -> Result<()> {
38    // A bad ref is reported under the error[CODE] contract every ref-taking
39    // command honours, before any other argument is looked at.
40    let parsed = match &ref_ {
41        Some(r) => Some(super::parse_ref_or_exit(r)?),
42        None => None,
43    };
44    let store_root = super::resolve_store_root()?;
45    let store = FsStore::new(store_root.clone())?;
46    let ctx = super::fetch::build_resolve_context()?;
47    let profile = CapabilityProfile::from_env()?;
48    match (parsed, file, from_dir) {
49        (Some(ref_), Some(f), None) => {
50            match add_user_pdf(&ref_, &f, force, &profile, &ctx, &store, &store_root).await {
51                Ok(o) => {
52                    print_err(format_args!(
53                        "added {} ({} bytes, user-supplied){} -> {}",
54                        ref_.as_input_str(),
55                        o.size_bytes,
56                        if o.replaced {
57                            ", replacing the stored PDF"
58                        } else {
59                            ""
60                        },
61                        o.path
62                    ));
63                    print_err(format_args!("     \"{}\"", o.title));
64                    Ok(())
65                }
66                Err(e) => fail(&e),
67            }
68        }
69        (None, None, Some(dir)) => {
70            run_from_dir(
71                &dir,
72                refs.as_deref(),
73                force,
74                apply,
75                &profile,
76                &ctx,
77                &store,
78                &store_root,
79            )
80            .await
81        }
82        _ => {
83            print_err(format_args!(
84                "error: give a ref and a file (`doiget add <ref> <file.pdf>`), or `--from-dir DIR`"
85            ));
86            Err(anyhow::Error::new(CliExit(2)))
87        }
88    }
89}

91fn fail(e: &AddError) -> Result<()> {
92    print_err(format_args!("error: {e}"));
93    // docs/ERRORS.md §4: a resolve failure exits as `fetch` would for its
94    // code, and 4 is the store's I/O failure (#649 review: these were
95    // swapped, 4 for a resolve and 1 for the store).
96    let code = match e {
97        AddError::Resolve { source, .. } => super::fetch::cli_exit_code(source.into()),
98        AddError::Store(_) | AddError::UnreadableEntry { .. } => 4,
99        _ => 2,
100    };
101    Err(anyhow::Error::new(CliExit(code)))
102}

104#[allow(clippy::too_many_arguments)]
105async fn run_from_dir(
106    dir: &Utf8Path,
107    refs: Option<&Utf8Path>,
108    force: bool,
109    apply: bool,
110    profile: &CapabilityProfile,
111    ctx: &doiget_core::source::FetchContext,
112    store: &FsStore,
113    store_root: &Utf8Path,
114) -> Result<()> {
115    let candidates = match refs {
116        Some(path) => {
117            let text = std::fs::read_to_string(path).with_context(|| format!("reading {path}"))?;
118            parse_input(&text, doiget_core::refs::Format::Auto, Some(path))
119                .into_iter()
120                .filter_map(Result::ok)
121                .map(|p| p.ref_)
122                .filter(|r| !has_pdf(store_root, &r.safekey()))
123                .collect::<Vec<_>>()
124        }
125        None => missing_in_store(store, store_root)?,
126    };
127    let mut files: Vec<Utf8PathBuf> = Vec::new();
128    for entry in std::fs::read_dir(dir).with_context(|| format!("reading {dir}"))? {
129        // A file the scan could not see would otherwise read as "not there"
130        // in the plan (#649 review), so each one is named.
131        let path = match entry {
132            Ok(e) => e.path(),
133            Err(err) => {
134                print_err(format_args!("warning: skipping an entry of {dir}: {err}"));
135                continue;
136            }
137        };
138        match Utf8PathBuf::from_path_buf(path) {
139            Ok(p) if p.extension().is_some_and(|x| x.eq_ignore_ascii_case("pdf")) => files.push(p),
140            Ok(_) => {}
141            Err(p) => print_err(format_args!(
142                "warning: skipping {} (the name is not UTF-8)",
143                p.display()
144            )),
145        }
146    }
147    files.sort();
148    if files.is_empty() {
149        bail!("no .pdf files in {dir}");
150    }
151
152    let mut plan: Vec<(Utf8PathBuf, Ref)> = Vec::new();
153    let mut unmatched: Vec<Utf8PathBuf> = Vec::new();
154    for f in files {
155        match match_file(&file_stem(&f), &candidates) {
156            Some(r) if !plan.iter().any(|(_, p)| p == r) => plan.push((f, r.clone())),
157            _ => unmatched.push(f),
158        }
159    }
160    let still_missing = candidates
161        .iter()
162        .filter(|r| !plan.iter().any(|(_, p)| p == *r))
163        .count();
164
165    let mut failures = 0usize;
166    for (f, r) in &plan {
167        if !apply {
168            print_err(format_args!("would add {} -> {}", f, r.as_input_str()));
169            continue;
170        }
171        match add_user_pdf(r, f, force, profile, ctx, store, store_root).await {
172            Ok(o) => print_err(format_args!(
173                "added {} -> {} ({})",
174                f,
175                r.as_input_str(),
176                o.path
177            )),
178            Err(e) => {
179                failures += 1;
180                print_err(format_args!("error: {f}: {e}"));
181            }
182        }
183    }
184    for f in &unmatched {
185        print_err(format_args!(
186            "unmatched: {f} (its name is not the id of exactly one entry missing a PDF; \
187             add it with `doiget add <ref> {f}`)"
188        ));
189    }
190    print_err(format_args!(
191        "add --from-dir: {} matched{}, {} unmatched file(s), {} entr{} still without a PDF",
192        plan.len(),
193        if apply {
194            ""
195        } else {
196            " (dry run: pass --apply to add them)"
197        },
198        unmatched.len(),
199        still_missing,
200        if still_missing == 1 { "y" } else { "ies" },
201    ));
202    if failures > 0 {
203        let code = i32::try_from(failures.min(255)).unwrap_or(255);
204        return Err(anyhow::Error::new(CliExit(code)));
205    }
206    Ok(())
207}

209fn has_pdf(store_root: &Utf8Path, key: &Safekey) -> bool {
210    store_root.join(format!("{}.pdf", key.as_str())).exists()
211}

213/// Store entries with metadata but no PDF, as refs.
214fn missing_in_store(store: &FsStore, store_root: &Utf8Path) -> Result<Vec<Ref>> {
215    let entries = blocking_section(|| store.list_recent(usize::MAX))?;
216    let mut out = Vec::new();
217    for e in entries {
218        if has_pdf(store_root, &e.safekey) {
219            continue;
220        }
221        let m = match blocking_section(|| store.read(&e.safekey)) {
222            Ok(Some(m)) => m,
223            Ok(None) => continue,
224            Err(err) => {
225                print_err(format_args!(
226                    "warning: skipping {} (its store entry could not be read: {err})",
227                    e.safekey.as_str()
228                ));
229                continue;
230            }
231        };
232        if let Some(d) = m.doi {
233            out.push(Ref::Doi(d));
234        } else if let Some(a) = m.arxiv_id {
235            out.push(Ref::Arxiv(a));
236        }
237    }
238    Ok(out)
