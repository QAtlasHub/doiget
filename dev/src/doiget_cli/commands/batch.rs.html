1//! `doiget batch <path>` — multi-ref orchestrator.
2//!
3//! Reads refs from a newline-separated text file, dispatches up to
4//! [`doiget_core::MAX_CONCURRENT_FETCHES`] in flight (the shared
5//! [`RateLimiter`](doiget_core::rate_limiter::RateLimiter) enforces the
6//! 5-per-second + 200 ms-per-source backoff invariants), and writes one
7//! shared provenance log file (`docs/PROVENANCE_LOG.md` §3).
8//!
9//! Spec:
10//! - `docs/PHASES.md` §4 success criterion: "`doiget batch <refs.txt>` honors
11//!   the rate cap and writes a hash-chained provenance log."
12//! - `crates/doiget-core/src/lib.rs` constants
13//!   [`doiget_core::MCP_BATCH_MAX_SIZE`] and
14//!   [`doiget_core::MAX_CONCURRENT_FETCHES`].
15//!
16//! ## Failure semantics
17//!
18//! - A file-read error (or a whole-input parse failure: `Decode` /
19//!   `UnsupportedFormat`) aborts the batch BEFORE any fetch. There is no
20//!   size cap: any number of refs is accepted and processed in bounded
21//!   windows of [`doiget_core::MCP_BATCH_MAX_SIZE`] (issue #304).
22//! - A single `Ref::parse` failure is non-fatal: the orchestrator emits a
23//!   `Resolve` row with `result=err` and continues with the remaining refs.
24//! - A single fetch failure is also non-fatal: per-task errors are recorded
25//!   by the `Source` impls (`Fetch` / `StoreWrite` rows) and counted by the
26//!   batch.
27//! - Exit code: `Ok(())` only when **every** parse + fetch succeeded; any
28//!   parse-error or per-ref-fetch-error returns `Err(...)` so the binary
29//!   surfaces a non-zero exit.
30//! - After the one-line count summary, a per-ref **failure digest**
31//!   (`<ref> -> <ERROR_CODE>`) is written to stderr (issue #222) so a
32//!   human / agent sees which refs failed and why without grepping the
33//!   JSONL provenance log. stdout stays clean (the `--mode json` JSON-Lines
34//!   is the machine channel).
35//! - The bookend rows are always emitted: one `SessionStart` at the top, one
36//!   `SessionEnd` at the bottom, regardless of per-ref outcome.

54/// One entry of the input file, on its way into the dispatch window.
55///
56/// `Rejected` carries the verdict the bibliography parser already reached.
57/// Before #500's reporting half this was a plain `Vec<String>`: every entry
58/// the parser could not turn into a `Ref` became a synthetic placeholder
59/// (`<unsupported-PMID:...>`, `<no-identifier:...>`) that was handed back to
60/// `Ref::parse`, whose only possible answer is `INVALID_REF`. So a PubMed
61/// `.bib` record with a valid `pmid` field was reported to the operator as a
62/// malformed reference -- the one claim #500 exists to stop doiget making --
63/// and the accompanying message described the placeholder rather than the
64/// entry. The verdict travels as a value now, so it cannot be flattened.
65enum BatchEntry {
66    /// A ref string still to be parsed.
67    Ref(String),
68    /// Unfetchable, with the reason already decided.
69    Rejected {
70        /// What to show in the JSONL `ref` field and the failure digest.
71        display: String,
72        /// The closed-set code the operator should see.
73        code: ErrorCode,
74        /// The parser's own message, about the entry rather than a
75        /// placeholder synthesised from it.
76        message: String,
77    },
78}

80/// Run the `doiget batch <path>` subcommand.
81///
82/// When `dry_run` is `true` (per ADR-0022 §1 + §3): read the input
83/// file, parse refs, and emit one `FetchPlan` JSON envelope line per ref
84/// on stdout. NO provenance log is opened, NO HTTP client is built, NO
85/// per-ref fetch runs, NO store write happens. Per-ref parse failures
86/// in dry-run mode are still counted and the function returns
87/// `Err(...)` if any ref failed to parse — the input was malformed and
88/// the caller should know.
89///
90/// When `dry_run` is `false`, runs the normal multi-ref orchestration —
91/// see the module-level docs for the failure-semantics contract.
92///
93/// # History
94///
95/// Slice 5 (PR #84 advisory item A2/A3 refactor): the previous
96/// `BatchOptions { dry_run: bool }` single-field option bundle plus the
97/// thin `run(path)` backwards-compat wrapper were collapsed into this
98/// single `dry_run: bool` parameter — the option bundle's single-bool
99/// shape was YAGNI, and the wrapper only existed to spare integration
100/// tests a `BatchOptions::default()` literal.
101pub async fn run_with_options(
102    path: String,
103    dry_run: bool,
104    only_failed: bool,
105    delay: Option<f64>,
106    user_agent: Option<String>,
107    mode: super::output::OutputMode,
108) -> Result<()> {
109    // `mode` honors ADR-0017. `Quiet` is a no-op here because batch's
110    // human summary already goes to stderr per ADR-0001. `Json` emits
111    // the ERRORS.md §3 CI-persona JSON-Lines per-ref shape on stdout
112    // (#205): one record per ref of the form
113    //   {"ok": true,  "ref": "..."}
114    //   {"ok": false, "ref": "...",
115    //    "error": {"code": "<ERROR_CODE>", "message": "..."}}
116    // The dry-run branch is unaffected — its product output is the
117    // FetchPlan envelope per ref, not a per-ref result record. A
118    // future follow-up will surface the full structured outcome
119    // (safekey / store_path / canonical_digest on success,
120    // `denial_context` on `CAPABILITY_DENIED`) once `fetch_one` returns
121    // the structured `FetchPaperOutcome` instead of `Result<()>`.
122    let json_mode = mode == super::output::OutputMode::Json;
123    // Step 1: read the input file. Failures surface before any fetch starts.
124    let raw =
125        std::fs::read_to_string(&path).with_context(|| format!("reading batch file: {path}"))?;
126
127    // Step 2: parse refs via the ADR-0030 bibliography adapter. The
128    // adapter auto-detects between plain refs / CSL-JSON / BibTeX
129    // (slice 2 follow-up) by path extension + content fingerprint,
130    // then yields one `Result<ParsedEntry, ParseError>` per
131    // discovered entry.
132    //
133    // To keep the rest of the pipeline byte-identical to the
134    // pre-ADR-0030 plain-refs flow, we materialise each entry into
135    // the same `inputs: Vec<String>` shape the downstream Step 7
136    // already drives. The Ok arm produces the canonical ref-input
137    // string; the per-entry `InvalidRef` arm produces the raw
138    // identifier verbatim so the downstream `Ref::parse` call still
139    // produces an `INVALID_REF` JSONL line with the offending
140    // string. Whole-input failures (`Decode` / `UnsupportedFormat`)
141    // abort before any fetch runs — the operator gets a single loud
142    // error rather than a silently-empty batch.
143    let path_utf8 = Utf8Path::new(&path);
144    let parsed = refs::parse_input(&raw, Format::Auto, Some(path_utf8));
145    let mut inputs: Vec<BatchEntry> = Vec::with_capacity(parsed.len());
146    for entry in parsed {
147        match entry {
148            Ok(p) => inputs.push(BatchEntry::Ref(p.ref_.as_input_str().to_string())),
149            // The raw identifier verbatim, so the downstream `Ref::parse`
150            // reports `INVALID_REF` against the string the user wrote. That
151            // IS the right claim here.
152            Err(ParseError::InvalidRef { raw, .. }) => inputs.push(BatchEntry::Ref(raw)),
153            // #500. The verdict travels as a value. It used to travel as a
154            // placeholder string handed back to `Ref::parse`, and `Ref::parse`
155            // has exactly one thing to say -- `INVALID_REF` -- so this arm and
156            // the `NoIdentifier` arm below reached the user as the SAME claim
157            // ("your bibliography is malformed") despite the comment here
158            // saying they differed, and the message described the synthetic
159            // placeholder rather than the user's entry. That claim is what
160            // #500 exists to stop doiget making about an entry that is fine.
161            Err(err @ ParseError::UnsupportedIdentifier { .. }) => {
162                let message = err.to_string();
163                let ParseError::UnsupportedIdentifier {
164                    kind,
165                    value,
166                    entry_key,
167                } = err
168                else {
169                    unreachable!("guarded by the pattern above")
170                };
171                let display = match entry_key {
172                    Some(k) => format!("<unsupported-{kind}:{value}:{k}>"),
173                    None => format!("<unsupported-{kind}:{value}>"),
174                };
175                inputs.push(BatchEntry::Rejected {
176                    display,
177                    // The input is valid and the support is absent. The same
178                    // code the MCP `batch_from_bibliography` and CLI `verify`
179                    // surfaces already gave this entry.
180                    code: ErrorCode::NotImplemented,
181                    message,
182                });
183            }
184            Err(err @ ParseError::NoIdentifier { .. }) => {
185                let message = err.to_string();
186                let ParseError::NoIdentifier { entry_key } = err else {
187                    unreachable!("guarded by the pattern above")
188                };
189                // The operator's citation key stays visible in the JSONL
190                // `ref` field. A future slice will plumb the structured
191                // `entry_key` into a dedicated error-object field
192                // (ADR-0030 §6).
193                let display = match entry_key {
194                    Some(k) => format!("<no-identifier:{k}>"),
195                    None => "<no-identifier>".to_string(),
196                };
197                inputs.push(BatchEntry::Rejected {
198                    display,
199                    code: ErrorCode::InvalidRef,
200                    message,
201                });
202            }
203            Err(ParseError::Decode { format, message }) => {
204                return Err(anyhow!("input did not deserialise as {format}: {message}"));
205            }
206            Err(ParseError::UnsupportedFormat { format }) => {
207                return Err(anyhow!(
208                    "input format '{format}' is not yet supported — \
209                     re-export your library as CSL-JSON or plain refs"
210                ));
211            }
212            // `ParseError` is `#[non_exhaustive]`; any future variant
213            // surfaces as a generic invalid-ref entry so the batch
214            // does not silently swallow a new failure class. The
215            // `tracing::error!` makes the unknown variant LOUD in
216            // logs so an operator notices when this arm fires —
217            // otherwise the operator would only see an `INVALID_REF`
218            // JSONL row indistinguishable from any other parse
219            // failure.
220            Err(other) => {
221                tracing::error!(
222                    error = %other,
223                    "encountered unknown ParseError variant; batch continues with placeholder \
224                     INVALID_REF — this should never happen on a current doiget-core build"
225                );
226                inputs.push(BatchEntry::Rejected {
227                    display: format!("<unhandled parse error: {other}>"),
228                    code: ErrorCode::InvalidRef,
229                    message: other.to_string(),
230                });
231            }
232        }
233    }
234
235    // Step 3: no hard cap. A local refs file is the user's whole working
236    // set, so the CLI `batch` processes ALL of it (issue #304): a previous
237    // hard `> MCP_BATCH_MAX_SIZE` abort lost the entire run — fetching
238    // nothing — the moment a bibliography crossed 100 refs, forcing a manual
239    // `split`. Instead `MCP_BATCH_MAX_SIZE` becomes the dispatch **window**
240    // (Step 7): refs are processed in chunks of that size so the number of
241    // concurrently-spawned tasks stays bounded, while the shared
242    // `RateLimiter` keeps the 5-per-second politeness invariant across every
243    // window. (`MCP_BATCH_MAX_SIZE` still hard-caps a single MCP
244    // `batch_fetch` request — that request-shape bound is unrelated to a
245    // local file the operator hands to the CLI.)
246
247    // Step 3a: dry-run branch (ADR-0022). Emit one `FetchPlan` envelope
248    // per ref on stdout WITHOUT opening the provenance log, building the
249    // HTTP client, or writing to the store. Per-ref parse failures still
250    // count toward the exit code so a malformed batch is visible.
251    if dry_run {
252        let store_root = resolve_store_root()?;
253        let mut parse_errors: usize = 0;
254        for entry in &inputs {
255            let input = match entry {
256                BatchEntry::Ref(input) => input,
257                BatchEntry::Rejected {
258                    display: shown,
259                    code,
260                    message,
261                } => {
262                    parse_errors += 1;
263                    tracing::warn!(
264                        input = %shown,
265                        error = %message,
266                        code = code.as_wire(),
267                        "skipping unfetchable batch entry in dry-run mode",
268                    );
269                    continue;
270                }
271            };
272            match Ref::parse(input) {
273                Ok(ref_) => {
274                    let plan = build_fetch_plan(&ref_, &store_root);
275                    emit_dry_run_plan_to_stdout(&ref_, &plan)?;
276                }
277                Err(e) => {
278                    parse_errors += 1;
279                    tracing::warn!(
280                        %input,
281                        error = %e,
282                        "skipping malformed batch entry in dry-run mode",
283                    );
284                }
285            }
286        }
287        if parse_errors > 0 {
288            return Err(anyhow!(
289                "dry-run batch: {} parse errors (no fetches attempted)",
290                parse_errors
291            ));
292        }
293        return Ok(());
294    }
295
296    // Step 4: build the harness once. All spawned tasks share an `Arc` to
297    // it; the harness already wraps the foundation modules in `Arc<...>` so
298    // this introduces no extra cloning overhead per task.
299    let harness = Arc::new(FetchHarness::from_env_with_ua(user_agent.as_deref())?);
300
301    // Step 5: SessionStart. Pass `None` for the ref — there is no single
302    // ref to attribute the session to.
303    harness.log_session_start(None)?;
304
305    // Step 6: bound concurrent in-flight tasks at
306    // `RateLimits::HARD_CODED.max_concurrent_fetches()` (= 5). The
307    // `RateLimiter` itself enforces the global rate + per-source backoff;
308    // this semaphore is purely the spawn-side cap on simultaneous tasks.
309    let max_concurrent = RateLimits::HARD_CODED.max_concurrent_fetches() as usize;
310    let semaphore = Arc::new(tokio::sync::Semaphore::new(max_concurrent));
311
312    // Step 7: dispatch in bounded windows. The ENTIRE input is processed —
313    // no ref is ever dropped (issue #304) — but at most `MCP_BATCH_MAX_SIZE`
314    // tasks exist at once: each window is spawned and then fully drained
315    // (Step 8) before the next window is spawned, so a large refs file never
316    // materialises one task per ref up front. The harness's shared
317    // `RateLimiter` enforces the 5-per-second + per-source backoff
318    // invariants across every window, so windowing changes only the
319    // spawn-side memory bound, never politeness. Counters accumulate across
320    // windows; parse errors are logged as `Resolve` rows off the shared log.
321    let mut parse_errors: usize = 0;
322    let mut fetch_ok: usize = 0;
323    let mut fetch_errors: usize = 0;
324    // `(ref, error_code)` per failed entry, drained into the end-of-batch
325    // stderr failure digest (issue #222).
326    let mut failures: Vec<(String, &'static str)> = Vec::new();
327    // Issue #324: count of refs skipped because a PDF already exists in the
328    // store (only populated when `--only-failed` is active).
329    let mut skipped: usize = 0;
330    let mut remaining = inputs.into_iter();
331    // Skip delay before the very first fetch; apply between all subsequent ones.
332    let mut is_first_spawn = true;
333    loop {
334        let window: Vec<BatchEntry> = remaining.by_ref().take(MCP_BATCH_MAX_SIZE).collect();
335        if window.is_empty() {
336            break;
337        }
338
339        let mut joins: tokio::task::JoinSet<TaskOutcome> = tokio::task::JoinSet::new();
340        for entry in window {
341            // Either the parser already reached a verdict about this entry, or
342            // `Ref::parse` reaches one now. Both land in the SAME reporting
343            // block below -- which is the point: a verdict that is not
344            // `INVALID_REF` now survives to the caller instead of being
345            // laundered through a placeholder string.
346            let parsed = match entry {
347                BatchEntry::Ref(input) => match Ref::parse(&input) {
348                    Ok(r) => Ok((input, r)),
349                    Err(e) => Err((input, ErrorCode::InvalidRef, e.to_string())),
350                },
351                BatchEntry::Rejected {
352                    display,
353                    code,
354                    message,
355                } => Err((display, code, message)),
356            };
357            let (input, ref_) = match parsed {
358                Ok(pair) => pair,
359                Err((input, code, message)) => {
360                    parse_errors += 1;
361                    failures.push((input.clone(), code.as_wire()));
362                    if json_mode {
363                        // #205: an unfetchable entry gets a JSONL line with the
364                        // human message in `error.message`. Per ERRORS.md §3.1
365                        // there is no `denial_context` on these -- the input
366                        // never reached a guard.
367                        emit_jsonl_failure(Some(&input), code.as_wire(), &message);
368                    }
369                    // Best-effort `Resolve` row capturing the parse failure;
370                    // we do NOT abort the batch on a single bad line.
371                    let _ = harness.log.append(RowInput {
372                        event: LogEvent::Resolve,
373                        result: LogResult::Err,
374                        capability: Capability::Oa,
375                        ref_: Some(&input),
376                        source: None,
377                        error_code: Some(code.as_wire()),
378                        size_bytes: None,
379                        license: None,
380                        store_path: None,
381                        // The input failed to parse as a Ref, so no
382                        // CanonicalRef can be minted (ADR-0021 §1 requires
383                        // a validated source_id).
384                        canonical_digest: None,
385                    });
386                    tracing::warn!(
387                        %input,
388                        error = %message,
389                        code = code.as_wire(),
390                        "skipping unfetchable batch entry",
391                    );
392                    continue;
393                }
394            };
395
396            // Issue #324: --only-failed — skip refs that already have a PDF
397            // in the store so a re-run only processes not-yet-succeeded entries.
398            // A PDF at `<store>/<safekey>.pdf` is the success signal; metadata-
399            // only entries (`.metadata/<safekey>.toml`) are NOT skipped because
400            // they indicate a NoOaUrl or prior Blocked outcome that may now
401            // succeed (e.g. after an allowlist update or preprint fallback).
402            if only_failed {
403                let safekey = ref_.safekey();
404                let pdf_path = harness
405                    .cfg
406                    .store_root
407                    .join(format!("{}.pdf", safekey.as_str()));
408                if pdf_path.exists() {
409                    tracing::debug!(
410                        ref_ = %input,
411                        path = %pdf_path,
412                        "already fetched; skipping (--only-failed)"
413                    );
414                    skipped += 1;
415                    if json_mode {
416                        emit_jsonl_already_fetched(&input);
417                    }
418                    continue;
419                }
420            }
421
422            let harness_task = Arc::clone(&harness);
423            let sem_task = Arc::clone(&semaphore);
424            if let Some(d) = delay {
425                if is_first_spawn {
426                    is_first_spawn = false;
427                } else {
428                    tokio::time::sleep(tokio::time::Duration::from_secs_f64(d)).await;
429                }
430            } else {
431                is_first_spawn = false;
432            }
433            joins.spawn(async move {
434                // `Semaphore::acquire_owned` only errors when the semaphore
435                // is closed; we never close it. The fallback maps that
436                // structurally-unreachable arm to a fetch failure rather
437                // than panicking.
438                let _permit = match sem_task.acquire_owned().await {
439                    Ok(p) => p,
440                    Err(_) => {
441                        // Map the semaphore-closed structural impossibility
442                        // to a typed `FetchError` (closest closed-set fit)
443                        // so `TaskOutcome::result` stays typed end-to-end.
444                        return TaskOutcome {
445                            input,
446                            result: Err(FetchError::SourceSchema {
447                                hint: "batch semaphore unexpectedly closed".to_string(),
448                            }),
449                        };
450                    }
451                };
452                let result = harness_task.fetch_one(&ref_).await;
453                TaskOutcome { input, result }
454            });
455        }
456
457        // Step 8: drain THIS window before spawning the next, so the
458        // in-flight task count stays bounded. Outcomes accumulate into the
459        // batch-wide counters; a single failure never aborts its siblings.
460        while let Some(joined) = joins.join_next().await {
461            let JoinedOutcome {
462                is_error,
463                json_record,
464                log_breadcrumb,
465                failure,
466            } = classify_joined(joined, json_mode);
467            if is_error {
468                fetch_errors += 1;
469            } else {
470                fetch_ok += 1;
471            }
472            if let Some(f) = failure {
473                failures.push(f);
474            }
475            if let Some(record) = json_record {
476                #[allow(clippy::print_stdout)]
477                {
478                    println!("{record}");
479                }
480            }
481            log_breadcrumb.emit();
482        }
483    }
484
485    let total_errors = parse_errors + fetch_errors;
486    let all_ok = total_errors == 0;
487
488    // Step 9: SessionEnd, always. Failure to append is best-effort; the
489    // caller already has whatever per-ref errors were observed.
490    // A batch bookend spans many refs, so there is no single terminal
491    // code to record; the per-ref rows carry those.
492    harness.log_session_end(all_ok, None, None);
493
494    // Step 10: stderr summary. ADR-0001: success / progress lines go to
495    // stderr; the workspace `print_stderr` lint is `warn`, promoted to deny
496    // in CI, so the localized `#[allow]` is the minimal intervention.
497    if skipped > 0 {
498        print_summary(format_args!(
499            "batch: {} OK, {} failed, {} skipped-already-fetched ({} parse errors, {} fetch errors)",
500            fetch_ok, total_errors, skipped, parse_errors, fetch_errors,
501        ));
502    } else {
503        print_summary(format_args!(
504            "batch: {} OK, {} failed ({} parse errors, {} fetch errors)",
505            fetch_ok, total_errors, parse_errors, fetch_errors,
506        ));
507    }
508
509    // Issue #222: a per-ref failure digest on stderr so a human / agent
510    // sees WHICH refs failed and WHY (the primary error code) without
511    // grepping the JSONL provenance log. stdout stays clean (ADR-0001 /
512    // #205 JSON-Lines is the machine channel).
513    if !failures.is_empty() {
514        print_summary(format_args!("batch failures ({}):", failures.len()));
515        for (ref_, code) in &failures {
516            print_summary(format_args!("  {ref_} -> {code}"));
517        }
518    }
519
520    if all_ok {
521        Ok(())
522    } else {
523        // Issue #143 / `docs/ERRORS.md` §4: the batch process exit code is
524        // the number of failures, capped at 255 (the dedicated "capped
525        // failure count for `batch`" exit). Previously a bare `anyhow!`
526        // fell through to `main`'s catch-all and exited 1 regardless of
527        // volume, so CI callers could not tell 1 failure from 200. The
528        // human-readable breakdown was ALREADY written to stderr by
529        // `print_summary` above, so `CliExit` carries only the code —
530        // `main` downcasts it exactly as it does for `fetch`'s `CliExit`.
531        let code = total_errors.min(255) as i32;
532        Err(anyhow::Error::new(CliExit(code)))
533    }
534}

536/// Per-task outcome carried out of the `JoinSet`. Holding the original input
537/// string lets the warn log breadcrumb the offending ref without re-parsing.
538///
539/// `Debug` is needed for the `classify_joined_panic_emits_null_ref_fetch_error`
540/// unit test (which spawns a panicking task and consumes the panic
541/// `Result<TaskOutcome, JoinError>` via `expect_err` — the latter requires `T: Debug`).
542#[derive(Debug)]
543struct TaskOutcome {
544    input: String,
545    result: Result<FetchPaperOutcome, FetchError>,
546}

548/// One-line summary written to stderr per ADR-0001 (stdio convention — the
549/// CLI never writes a success line to stdout). `print_stderr` is a workspace
550/// `warn` promoted to `deny` under `-D warnings`; the localized `#[allow]`
551/// pinpoints the one intentional eprintln.
552#[allow(clippy::print_stderr)]
553fn print_summary(args: std::fmt::Arguments<'_>) {
554    eprintln!("{args}");
555}

557/// Outcome of a single `JoinSet::join_next()` result, classified per
558/// #205 (success / fetch error / task panic) plus the JSON-Lines
559/// record (if `json_mode`) and the breadcrumb level for the tracing
560/// log. Extracted from the drain loop so it can be unit-tested without
561/// running the full orchestrator (self-review for #209 §7 +
562/// codecov/patch coverage closure).
563struct JoinedOutcome {
564    /// True when this entry contributes to `fetch_errors`.
565    is_error: bool,
566    /// The JSONL record to emit, when `json_mode` is on. `None` means
567    /// "no record" (either `json_mode` was false, or the variant has
568    /// no record by design — currently every variant produces a
569    /// record under json_mode, but the option leaves room for one).
570    json_record: Option<serde_json::Value>,
571    /// Source-side breadcrumb for the per-ref tracing line, captured
572    /// here so `classify_joined` stays pure and the loop just calls
573    /// `.log()` on the outcome.
574    log_breadcrumb: LogBreadcrumb,
575    /// `(ref, error_code)` for a failed entry, collected into the
576    /// end-of-batch stderr failure digest (issue #222) so a human / agent
577    /// sees WHICH refs failed and WHY without grepping the JSONL log.
578    /// `None` on success. The ref is a sentinel only for a task panic
579    /// (the `JoinSet` lost the input).
580    failure: Option<(String, &'static str)>,
581}
582
583enum LogBreadcrumb {
584    /// Success — no per-ref tracing line.
585    None,
586    /// `tracing::warn!(input, error)` for a normal fetch failure.
587    FetchFailed { input: String, error_dbg: String },
588    /// `tracing::error!(error)` for a task panic / cancellation.
589    TaskPanicked { error_dbg: String },
590}
591
592impl LogBreadcrumb {
593    /// Emit the per-ref breadcrumb to the tracing subscriber. Pulled
594    /// out of `classify_joined` so the helper stays pure and out of
595    /// the drain-loop body so an owned `JoinedOutcome` can be
596    /// destructured without borrow-check trouble.
597    fn emit(self) {
598        match self {
599            LogBreadcrumb::None => {}
600            LogBreadcrumb::FetchFailed { input, error_dbg } => {
601                tracing::warn!(%input, %error_dbg, "batch entry fetch failed");
602            }
603            LogBreadcrumb::TaskPanicked { error_dbg } => {
604                tracing::error!(%error_dbg, "batch task panicked or was cancelled");
605            }
606        }
607    }
608}
609
610/// Classify a single `JoinSet::join_next()` result. Pure function
611/// (no I/O, no tracing) — `JoinedOutcome::log` handles the breadcrumb
612/// side-effect at the drain site. Unit-tested below for every variant
613/// including a real `JoinError` synthesised by spawning a panicking
614/// task on a tiny `JoinSet<()>`.
615fn classify_joined(
616    joined: Result<TaskOutcome, tokio::task::JoinError>,
617    json_mode: bool,
618) -> JoinedOutcome {
619    match joined {
620        Ok(TaskOutcome { input, result }) => match result {
621            Ok(outcome) => {
622                // #210 / `docs/ERRORS.md` §3+§6: a `Blocked` PDF leg
623                // is NOT a clean success even though the typed Result
624                // is `Ok` — surface it as a structured failure record
625                // with the `denial_context` the orchestrator carried
626                // on `PdfLegStatus::Blocked.denial`.
627                if let PdfLegStatus::Blocked {
628                    code,
629                    message,
630                    denial,
631                    ..
632                } = &outcome.pdf_leg
633                {
634                    let effective = effective_blocked_code(*code, denial.as_ref());
635                    let record = json_mode.then(|| {
636                        build_jsonl_failure(
637                            Some(&input),
638                            effective.as_wire(),
639                            message,
640                            denial.as_ref(),
641                            &outcome.attempts,
642                        )
643                    });
644                    let error_dbg =
645                        format!("pdf_leg=Blocked code={effective:?} message={message:?}");
646                    let failure = Some((input.clone(), effective.as_wire()));
647                    return JoinedOutcome {
648                        is_error: true,
649                        json_record: record,
650                        log_breadcrumb: LogBreadcrumb::FetchFailed { input, error_dbg },
651                        failure,
652                    };
653                }
654                let result_value = json_mode.then(|| outcome_to_result_value(&outcome));
655                JoinedOutcome {
656                    is_error: false,
657                    json_record: json_mode
658                        .then(|| build_jsonl_success(&input, result_value.clone().flatten())),
659                    log_breadcrumb: LogBreadcrumb::None,
660                    failure: None,
661                }
662            }
663            Err(e) => {
664                let error_dbg = format!("{e:?}");
665                let json_msg = format!("{e}");
666                let code: ErrorCode = (&e).into();
667                let denial = Option::<DenialContext>::from(&e);
668                let record = json_mode.then(|| {
669                    build_jsonl_failure(
670                        Some(&input),
671                        code.as_wire(),
672                        &json_msg,
673                        denial.as_ref(),
674                        // A typed `FetchError` carries no trace — the
675                        // orchestrator's `attempts` live on the outcome,
676                        // which this arm never got.
677                        &[],
678                    )
679                });
680                let failure = Some((input.clone(), code.as_wire()));
681                JoinedOutcome {
682                    is_error: true,
683                    json_record: record,
684                    log_breadcrumb: LogBreadcrumb::FetchFailed { input, error_dbg },
685                    failure,
686                }
687            }
688        },
689        Err(join_err) => {
690            let error_dbg = format!("{join_err:?}");
691            // The JoinSet has lost the originating input by the
692            // time the panic surfaces. Honest serialisation:
693            // `"ref": null` instead of a `"<task-panic>"`
694            // sentinel that a consumer doing `retry(rec["ref"])`
695            // would mishandle. Self-review for #209 §1.
696            let json_msg = format!("batch task panicked: {join_err}");
697            let record =
698                json_mode.then(|| build_jsonl_failure(None, "FETCH_ERROR", &json_msg, None, &[]));
699            JoinedOutcome {
700                is_error: true,
701                json_record: record,
702                log_breadcrumb: LogBreadcrumb::TaskPanicked { error_dbg },
703                failure: Some(("(task panicked — ref lost)".to_string(), "FETCH_ERROR")),
704            }
705        }
706    }
707}

709/// Build the `result` sub-object for the JSONL success record per
710/// ERRORS.md §3: `{safekey, store_path, canonical_digest}`. Returns
711/// `None` if any field cannot be serialised; the caller falls back to
712/// the bare `{ok, ref}` shape so a serialisation glitch never drops
713/// the success signal.
714fn outcome_to_result_value(outcome: &FetchPaperOutcome) -> Option<serde_json::Value> {
715    Some(serde_json::json!({
716        "safekey":          outcome.safekey,
717        "store_path":       outcome.path.as_str(),
718        "canonical_digest": outcome.canonical_digest,
719        // OA transparency (#281 item 4): gold/green/hybrid/bronze/closed,
720        // or null when not determined.
721        "oa_status":        outcome.oa_status,
722    }))
723}

725/// Serialise a `DenialContext` to a JSON value for the `error.denial_context`
726/// field. `DenialContext: serde::Serialize` already; returns `None` on
727/// the (today unreachable) serialise-failure path rather than aborting
728/// the record.
729fn denial_context_to_value(dc: &DenialContext) -> Option<serde_json::Value> {
730    serde_json::to_value(dc).ok()
731}

733/// #205 + #210: build the JSON-Lines success record value for
734/// `ref_input`. Per ERRORS.md §3, the wire shape is
735/// `{"ok": true, "ref": "...", "result": {...}}` when `result` is
736/// `Some`, falling back to `{"ok": true, "ref": "..."}` when `result`
737/// is `None` (the latter is reserved for paths that don't carry a
738/// structured outcome — today every fetch path does, so `result` is
739/// `Some` in practice; the `None` arm preserves the unit-test path
740/// that asserts shape without constructing a full `FetchPaperOutcome`).
741///
742/// Returned as `serde_json::Value` so unit tests can assert the shape
743/// without a stdout-capture dance; the integration site wraps it with
744/// `println!`.
745fn build_jsonl_success(ref_input: &str, result: Option<serde_json::Value>) -> serde_json::Value {
746    match result {
747        Some(r) => serde_json::json!({ "ok": true, "ref": ref_input, "result": r }),
748        None => serde_json::json!({ "ok": true, "ref": ref_input }),
749    }
750}

752/// #205 + #210: build the JSON-Lines failure record value with `code`,
753/// `message`, and an optional `denial_context` (ADR-0023). The wire
754/// shape is `{"ok": false, "ref": "<ref>"|null, "error": {"code": "...",
755/// "message": "..."[, "denial_context": {...}]}}`.
756///
757/// `denial_context` is omitted from the record entirely when `None` —
758/// fields are not present rather than `null` — so consumers can use the
759/// `denial_context` key's presence as a discriminator between a typed
760/// policy denial and a bare transport / config error.
761///
762/// `ref_input` is `Option<&str>` because the JoinSet panic arm has lost
763/// the originating input by the time the panic surfaces: serialising
764/// `null` is more honest than a sentinel string like `"<task-panic>"`
765/// (which a consumer doing `retry(rec["ref"])` would mishandle by
766/// trying to refetch the literal sentinel — self-review for #209 §1).
767fn build_jsonl_failure(
768    ref_input: Option<&str>,
769    code: &str,
770    message: &str,
771    denial: Option<&DenialContext>,
772    attempts: &[doiget_core::orchestrator::SourceAttempt],
773) -> serde_json::Value {
774    let mut error = serde_json::json!({ "code": code, "message": message });
775    if let Some(obj) = error.as_object_mut() {
776        if let Some(dc) = denial {
777            // Route through the logged helper (#154): a bare `json!` would
778            // coerce a future serialization failure to `null` with no trace.
779            if let Some(v) = denial_context_to_value(dc) {
780                obj.insert("denial_context".to_string(), v);
781            }
782            // #459: `denial_context` says what was refused; this says what
783            // to do. A CI consumer reading only the first can report a
784            // recoverable failure as a permanent one.
785            let r = doiget_core::remediation::for_denial(dc);
786            if !r.is_empty() {
787                obj.insert("remediation".to_string(), serde_json::json!(r));
788            }
789        }
790        // #459: the #438 resolution trace, which until now existed only as
791        // CLI text. Omitted entirely when empty rather than emitted as
792        // `[]`, so "no trace available" and "the trace is empty" stay
793        // distinguishable.
794        if !attempts.is_empty() {
795            obj.insert(
796                "attempts".to_string(),
797                doiget_core::orchestrator::attempts_to_value(attempts),
798            );
799        }
800    }
801    serde_json::json!({
802        "ok":    false,
803        "ref":   ref_input,
804        "error": error,
805    })
806}

808/// `emit_jsonl_failure` is still called from the parse-failure site
809/// (Step 7's `Ref::parse` branch); the JoinSet-drain site now uses
810/// [`classify_joined`] + an inline `println!`, so the matching
811/// `emit_jsonl_success` thin wrapper is no longer needed and was
812/// removed alongside the drain refactor. The `denial_context` slot is
813/// always `None` here — `INVALID_REF` is a pre-guard parse error, not
814/// a policy denial (`docs/ERRORS.md` §3.1).
815#[allow(clippy::print_stdout)]
816fn emit_jsonl_failure(ref_input: Option<&str>, code: &str, message: &str) {
817    println!(
818        "{}",
819        build_jsonl_failure(ref_input, code, message, None, &[])
820    );
821}

823/// Emit a JSON-Lines record for a ref skipped because its PDF already exists
824/// in the store (`--only-failed`, issue #324). Uses `ok: true` + an
825/// `already_fetched: true` extension field so downstream consumers can
826/// distinguish "new success" from "pre-existing success without re-fetch".

835}

1308    #[test]
1309    fn over_limit_input_is_windowed_not_rejected() {
1310        // Issue #304: an input above `MCP_BATCH_MAX_SIZE` is no longer
1311        // rejected — it is dispatched in bounded windows of that size, and
1312        // every ref is processed. This pins the windowing arithmetic that
1313        // the dispatch loop relies on (`into_iter().take(MAX)` until empty):
1314        // `n` refs split into `ceil(n / MAX)` windows, the last possibly
1315        // short, with no ref lost.
1316        let n = MCP_BATCH_MAX_SIZE + 1;
1317        let inputs: Vec<String> = (0..n)
1318            .map(|i| format!("arxiv:2401.{:05}", 10000 + i))
1319            .collect();
1320
1321        let mut remaining = inputs.clone().into_iter();
1322        let mut windows: Vec<usize> = Vec::new();
1323        loop {
1324            let window: Vec<String> = remaining.by_ref().take(MCP_BATCH_MAX_SIZE).collect();
1325            if window.is_empty() {
1326                break;
1327            }
1328            windows.push(window.len());
1329        }
1330
1331        // ceil(n / MAX) windows; every ref accounted for; each window within
1332        // the bound.
1333        assert_eq!(windows.len(), n.div_ceil(MCP_BATCH_MAX_SIZE));
1334        assert_eq!(windows.iter().sum::<usize>(), n);
1335        assert!(windows.iter().all(|&w| w <= MCP_BATCH_MAX_SIZE));
1336        assert_eq!(*windows.last().unwrap(), n % MCP_BATCH_MAX_SIZE);
1337    }
